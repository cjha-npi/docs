27 /**
28 * \brief Extends QCompleter to show popup when QLineEdit gains focus or is empty.
29 * \details Extends the standard QCompleter behavior by providing completion
30 * popups in two specific cases not handled by Qt's normal completion mechanism:
31 * - When `QLineEdit` gains focus, and
32 * - When Backspace/Delete is used and `QLineEdit` text is empty.
33 *
34 * Focus caused by a popup closing or by application-window activation is
35 * ignored to prevent the completion popup from immediately reopening during
36 * normal popup dismissal or window focus changes.
37 *
38 * The additional behavior is applied only when the completer's associated
39 * widget is a QLineEdit. Standard QCompleter event handling is otherwise
40 * preserved.
41 */
42 class EmptyCompleter final : public QCompleter {
43 protected:
44 /// \brief Handles focus in and Backspace/Delete key press to show completion popup.
45 /// \param watched Object receiving the event.
46 /// \param event Event being processed.
47 /// \return Result of the standard QCompleter event handling.
48 bool eventFilter(QObject* watched, QEvent* event) override
49 {
50 // Preserve standard completer handling before applying the additional behavior.
51 const bool handled = QCompleter::eventFilter(watched, event);
52
53 if(event->type() == QEvent::FocusIn) {
54 // watched is the associated QLineEdit
55 if(watched == widget()) {
56 // Ignore focus returned from a popup or caused by application-window activation.
57 // Not ignoring Qt::PopupFocusReason can cause infinite loop.
58 if(const auto* focusEvent = static_cast<QFocusEvent*>(event);
59 focusEvent->reason() != Qt::PopupFocusReason &&
60 focusEvent->reason() != Qt::ActiveWindowFocusReason)
61 showCompletion_();
62 }
63 }
64 else if(event->type() == QEvent::KeyPress) {
65 // Refresh completion when text may have been removed.
66 // A QCompleter consumes the key press event on the associated
67 // QLineEdit as long as it is within the completer's range.
68 // That's why we are not checking for watched being QLineEdit.
69 if(const auto* keyEvent = static_cast<QKeyEvent*>(event);
70 keyEvent->key() == Qt::Key_Backspace ||
71 keyEvent->key() == Qt::Key_Delete) {
72 // Check based on timer because Qt clears the text after
73 // this event has been processed.
74 QTimer::singleShot(0, this, [this] { showCompletion_(); });
75 }
76 }
77
78 return handled;
79 }
80
81 private:
82 /// \brief Updates the completion prefix and displays the completion popup.
83 /// \details Uses the current text of the associated QLineEdit as the
84 /// completion prefix. An empty string is therefore a valid prefix
85 /// and allows the completer to display all available completions.
86 /// No action is performed if the associated widget is not a QLineEdit
87 /// or if the line edit does not currently have focus.
88 void showCompletion_()
89 {
90 auto* lineEdit = qobject_cast<QLineEdit*>(widget());
91 if(!lineEdit || !lineEdit->hasFocus()) return;
92 setCompletionPrefix(lineEdit->text());
93 complete();
94 }
95 };
