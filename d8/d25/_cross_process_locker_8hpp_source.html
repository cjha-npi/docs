1/**
2 * \file CrossProcessLocker.hpp
3 * \brief RAII locker for a session-local Windows named mutex.
4 * \details
5 * Defines `npi::app::CrossProcessLocker`, a small RAII helper that creates or opens
6 * a Windows named mutex, waits for ownership, releases ownership, and closes
7 * the mutex handle.
8 *
9 * A named mutex can synchronize access between:
10 * - Multiple threads within one process.
11 * - Multiple processes running in the same Windows session.
12 *
13 * Scope:
14 * - The constructor prepends "Local\\" to the provided name, limiting
15 * visibility to the current Windows session.
16 * - A "Global\\" prefix would place the mutex in the global namespace,
17 * subject to the mutex object's access-control settings.
18 *
19 * Ownership:
20 * - Successful construction guarantees ownership of the mutex.
21 * - Invalid input or any mutex creation/acquisition failure terminates the
22 * application through an out-of-process stop-message helper.
23 * - Destruction releases ownership and closes the mutex handle.
24 *
25 * Abandoned mutexes:
26 * - WAIT_ABANDONED is treated as successful acquisition, as required by
27 * Windows mutex semantics.
28 * - A warning is logged because the protected resource may be inconsistent.
29 * - `wasAbandoned_()` allows the caller to detect this condition and perform
30 * any required validation or recovery.
31 *
32 * \warning
33 * Invalid input or failure to create or acquire the mutex causes immediate
34 * application termination.
35 */

62 /**
63 * \class CrossProcessLocker
64 * \brief Scoped owner of a session-local Windows named mutex.
65 * \details
66 * Creates or opens a named mutex, waits indefinitely to acquire ownership,
67 * and releases ownership during destruction. The mutex name is prefixed with
68 * "Local\\", limiting synchronization to the current Windows session.
69 *
70 * Copying and moving are disabled so that exactly one locker object owns and
71 * releases the acquired mutex handle.
72 *
73 * WAIT_ABANDONED is treated as successful acquisition. In that case,
74 * `wasAbandoned_()` returns true so the caller can validate or repair the
75 * protected resource when necessary.
76 *
77 * \warning
78 * Invalid input or failure to create or acquire the mutex causes immediate
79 * application termination.
80 */
81 class CrossProcessLocker final {
82 public:
83 /// \brief Create or open a session-local named mutex and acquire it.
84 /// \param name Logical mutex name without "Local\\" and without backslashes. Example: "npi-SharedFileLock".
85 /// \details
86 /// Internally, the name is converted to UTF-16 and prefixed with "Local\\". This constructor:
87 /// 1) Calls CreateMutexW(nullptr, FALSE, L"Local\\{name}").
88 /// Passing FALSE means the mutex is not acquired at creation time.
89 /// 2) If the handle is valid, calls WaitForSingleObject(h, INFINITE).
90 /// On WAIT_OBJECT_0 or WAIT_ABANDONED, the lock is considered acquired.
91 /// If another thread/process already owns the mutex, this call blocks until it becomes available.
92 explicit CrossProcessLocker(const QString& name)
93 {
94 // Trim the name only for validation; preserve the original name for the mutex.
95 const QString cleaned = name.trimmed();
96
97 if(cleaned.isEmpty()) {
98 npi::app::winStopMsgExe(
99 QStringLiteral(
100 "%1 Empty Name!\n\n"
101 "A locker object cannot be created with an empty or "
102 "whitespace-only name.\n\n"
103 "The application cannot continue without the locker "
104 "object and will now exit.")
105 .arg(ErrMsgPrefix_.toString()));
106 }
107
108 if(cleaned.contains('\\')) {
109 npi::app::winStopMsgExe(
110 QStringLiteral(
111 "%1 Invalid Name!\n\n"
112 "'%2' is an invalid locker-object name. Windows named "
113 "mutexes do not allow names containing '\\'.\n\n"
114 "The application cannot continue without the locker "
115 "object and will now exit.")
116 .arg(ErrMsgPrefix_.toString(), name));
117 }
118
119 const std::wstring lockName = (QStringLiteral("Local\\") + name).toStdWString();
120
121 // Obtain a handle to the mutex object, it does not lock it. If we pass here
122 // TRUE instead of FALSE then it locks but only when the mutex is newly created.
123 // If the named mutex already exists, then passing TRUE instead of FALSE will
124 // result in ownership request being ignored and so no lock.
125 const HANDLE handle = ::CreateMutexW(nullptr, FALSE, lockName.c_str());
126
127 // CreateMutexW returns nullptr if it is unable to create the mutex handle.
128 if(handle == nullptr) {
129 const DWORD error = ::GetLastError();
130 npi::app::winStopMsgExe(
131 QStringLiteral(
132 "%1 Unable to Create Mutex!\n\n"
133 "CreateMutexW failed for '%2'. The application cannot "
134 "continue without the locker object and will now exit.\n\n"
135 "Error: %3")
136 .arg(ErrMsgPrefix_.toString(), name, npi::ind::winErrorText(error)));
137 }
138
139 // Acquire the lock, we wait for as long as needed.
140 const DWORD waitResult = ::WaitForSingleObject(handle, INFINITE);
141
142 // The mutex was available and acquired.
143 if(waitResult == WAIT_OBJECT_0) {
144 handle_ = handle;
145 return;
146 }
147
148 // The previous owning thread or process terminated without releasing it.
149 // We have acquired the mutex, but the protected data may be inconsistent.
150 // We treat it as success but provide abandoned info to caller, if inquired.
151 if(waitResult == WAIT_ABANDONED) {
152 handle_ = handle;
153 abandoned_ = true;
154
155 qWarning().noquote()
156 << QStringLiteral(
157 "%1 Mutex '%2' was abandoned by its previous owner. "
158 "Ownership was acquired, but the protected data "
159 "may be inconsistent.").arg(ErrMsgPrefix_.toString(), name);
160
161 return;
162 }
163
164 // Preserve the wait error before calling another Windows API function.
165 const DWORD error = waitResult == WAIT_FAILED
166 ? ::GetLastError()
167 : ERROR_GEN_FAILURE;
168
169 // Close the local handle.
170 ::CloseHandle(handle);
171
172 // Failure to acquire the mutex, we show it and terminate the application.
173 npi::app::winStopMsgExe(
174 QStringLiteral(
175 "%1 Unable to Acquire Mutex!\n\n"
176 "Waiting for mutex '%2' failed. The application cannot "
177 "continue without the locker object and will now exit.\n\n"
178 "Error: %3")
179 .arg(ErrMsgPrefix_.toString(), name, npi::ind::winErrorText(error)));
180 }
181
182 /// \brief Release ownership of the mutex and close its handle.
183 /// \details
184 /// This class provides scoped RAII ownership and therefore has no explicit
185 /// lock or unlock member functions. The mutex remains owned throughout the
186 /// lifetime of the locker and is released during destruction.
187 ~CrossProcessLocker() noexcept
188 {
189 ::ReleaseMutex(handle_);
190 ::CloseHandle(handle_);
191 }
192
193 CrossProcessLocker(CrossProcessLocker&& other) = delete; ///< Deleted Move Constructor
194 CrossProcessLocker(const CrossProcessLocker& other) = delete; ///< Deleted Copy Constructor
195 CrossProcessLocker& operator=(CrossProcessLocker&& other) = delete; ///< Deleted Move Assignment
196 CrossProcessLocker& operator=(const CrossProcessLocker& other) = delete; ///< Deleted Copy Assignment
197
198 /// \brief Function to query if the mutex was abandoned by its previous owner.
199 /// \return true if the mutex was abandoned by its previous owner; otherwise false.
200 /// \details This function is provided for convenience, so that the caller
201 /// can check if the previous owner did not release the mutex correctly
202 /// and perform any checks or cleanups as required for the shared resource.
203 [[nodiscard]] bool wasAbandoned_() const noexcept
204 {
205 return abandoned_;
206 }
207
208 private:
209 /// \brief Name for log and popup messages.
210 static constexpr QLatin1StringView ErrMsgPrefix_{"Cross Process Locker:"};
211
212 HANDLE handle_{nullptr}; ///< Handle to the acquired named mutex.
213 bool abandoned_{false}; ///< Whether acquisition returned WAIT_ABANDONED.
214 };
