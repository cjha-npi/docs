27 /**
28 * \brief Helper for `StageWindow` class. A RAII guard that restores a widget's window opacity on scope exit.
29 * \details For use when the target widget's opacity is set to 0.0 before
30 * an animation is fully wired. If control leaves the scope early (exception
31 * or early return) and the guard is still active, it restores the widget to
32 * full opacity (1.0) in the destructor. After the animation has been started
33 * and its own finished-handler will handle restoration, call release_() to
34 * disable the guard.
35 * \note
36 * The guard holds a QPointer to avoid dereferencing a deleted widget. If the
37 * widget is destroyed before scope exit, no action is taken.
38 * \warning
39 * Construct and use from the GUI thread only. The opacity change affects GUI state.
40 */
41 class OpacityGuard {
42 public:
43 OpacityGuard() = delete; ///< deleted default constructor
44 OpacityGuard(const OpacityGuard&) = delete; ///< deleted copy constructor
45 OpacityGuard& operator=(const OpacityGuard&) = delete; ///< deleted copy assignment
46 OpacityGuard(OpacityGuard&&) = delete; ///< deleted move constructor
47 OpacityGuard& operator=(OpacityGuard&&) = delete; ///< deleted move assignment
48
49 /// \brief Construct a guard for the given widget.
50 /// \param w Pointer to the widget whose opacity should be restored on scope exit.
51 explicit OpacityGuard(QWidget* w) : guardedWidget_{w} {}
52
53 /// \brief Destructor; restores opacity if still active and the widget is alive.
54 ~OpacityGuard()
55 {
56 if(active_ && !guardedWidget_.isNull())
57 guardedWidget_->setWindowOpacity(1.0);
58 }
59
60 /// \brief Disable the guard so the destructor does not restore opacity.
61 /// \details Call after successfully starting the animation and arranging
62 /// for normal restoration (e.g., in the animation's finished handler).
63 /// \post active_ is false; the destructor will not modify widget opacity.
64 void release_() { active_ = false; }
65
66 private:
67 bool active_{true}; ///< Active state of the guard
68 QPointer<QWidget> guardedWidget_; ///< QPointer to the guarded widget
69 };
70}

126void npi::ind::StageWindow::animate_() const
127{
128 if(widget_.isNull() || !widget_->isWindow())
129 return;
130
131 // Determine starting rect for the overlay animation.
132 const QRect begRect{
133 caller_.isNull()
134 ? QRect{QCursor::pos().x() - 1, QCursor::pos().y() - 1, 3, 3}
135 : (caller_->isWindow()
136 ? caller_->frameGeometry()
137 : QRect{caller_->mapToGlobal(QPoint{0, 0}), caller_->size()})
138 };
139
140 // Position target and compute its final frame geometry now.
141 positionWidget_();
142
143 // Make target invisible first, then show it correctly so frameGeometry() is valid for the end rect.
144 widget_->setWindowOpacity(0.0);
145 callWidgetCorrectShowMethod_();
146
147 // Guard ensures we do not leave the widget transparent on early exit.
148 OpacityGuard guard{widget_};
149
150 // Derive overlay background color for a subtle contrast flash.
151 QColor bgColor{widget_->palette().color(QPalette::Window)};
152 const int avg = (bgColor.red() + bgColor.green() + bgColor.blue()) / 3;
153 bgColor = (avg > 127) ? bgColor.darker(120) : bgColor.lighter(120);
154
155 // Create temporary overlay; no parent so it is its own top-level.
156 QWidget* temp = new QWidget;
157 temp->setWindowFlags(Qt::FramelessWindowHint | Qt::Tool);
158 temp->setAttribute(Qt::WA_StyledBackground, true);
159 temp->setAttribute(Qt::WA_TransparentForMouseEvents, true);
160 temp->setGeometry(begRect);
161 temp->setStyleSheet(QStringLiteral("background-color: %1;").arg(bgColor.name()));
162
163 // Determine duration.
164 int dura = duration_;
165 if(dura == -1) { // Use auto-duration heuristic
166 const QPoint diff = begRect.topLeft() - widget_->frameGeometry().topLeft();
167 // Heuristic: base plus path-length factor, clamped to [MinDura_, MaxDura_].
168 const int term = static_cast<int>(qSqrt((diff.x() * diff.x()) + (diff.y() * diff.y())) / 4);
169 dura = qMin(MaxDura_, MinDura_ + term);
170 }
171
172 // Animate overlay geometry.
173 QPropertyAnimation* amin = new QPropertyAnimation(temp, "geometry");
174 amin->setDuration(dura);
175 amin->setStartValue(begRect);
176 amin->setEndValue(widget_->frameGeometry());
177 amin->setEasingCurve(easingCurve_);
178
179 // Capture a safe handle to the widget to avoid dereferencing a dead `this`.
180 QPointer<QWidget> target = widget_;
181
182 QObject::connect(amin, &QPropertyAnimation::finished, [temp, target]() {
183 if(!target.isNull())
184 target->setWindowOpacity(1.0);
185 temp->deleteLater();
186 });
187
188 temp->show();
189 amin->start(QAbstractAnimation::DeleteWhenStopped);
190
191 // At this point the normal path hands off restoration to the finished-callback.
192 // Disable the guard so it does not undo the fade.
193 guard.release_();
194}

196void npi::ind::StageWindow::positionWidget_() const
197{
198 if(widget_.isNull() || !widget_->isWindow())
199 return;
200
201 // Calculate the extra space needed for the title bar.
202 // If the widget has a title (WindowTitleHint is set), obtain the title bar height
203 const auto titleHeight{(widget_->windowFlags() & Qt::WindowTitleHint) ? qApp->style()->pixelMetric(QStyle::PM_TitleBarHeight) : 0};
204
205 // Calculate the extra space needed for the window frame.
206 // If the widget is frameless, set frameSize to 0; otherwise, use the default frame width.
207 const auto frameSize{(widget_->windowFlags() & Qt::FramelessWindowHint) ? 0 : qApp->style()->pixelMetric(QStyle::PM_DefaultFrameWidth)};
208
209 // Determine full framed size.
210 QSize widFullSize;
211 if(widget_->geometry() == widget_->frameGeometry()) {
212 // Not shown yet; synthesize framed size from client size.
213 widFullSize = QSize{
214 widget_->size().width() + (2 * frameSize),
215 widget_->size().height() + titleHeight + (2 * frameSize)
216 };
217 }
218 else {
219 widFullSize = widget_->frameGeometry().size();
220 }
221
222 // Start from current position.
223 QPoint widPos{widget_->pos()};
224
225 // Choose a screen reference.
226 QScreen* screen{qApp->screenAt(widget_->frameGeometry().center())};
227
228 if(!center_.isNull()) {
229 // Center over `center_`.
230 screen = qApp->screenAt(center_->geometry().center());
231 const QRect cenRect = center_->isWindow() ? center_->frameGeometry() : center_->geometry();
232 const int widPosX = cenRect.x() + (cenRect.width() - widFullSize.width()) / 2;
233 const int widPosY = cenRect.y() + (cenRect.height() - widFullSize.height()) / 2;
234 widPos = QPoint{widPosX, widPosY};
235 }
236 else if(widget_->pos() == QPoint{0, 0} && widget_->geometry() == widget_->frameGeometry()) {
237 // Likely first show and no explicit centering widget; center on cursor's screen.
238 screen = qApp->screenAt(QCursor::pos());
239 if(screen) {
240 const QRect avail = screen->availableGeometry();
241 const int widPosX = avail.center().x() - (widFullSize.width() / 2);
242 const int widPosY = avail.center().y() - (widFullSize.height() / 2);
243 widPos = QPoint{widPosX, widPosY};
244 }
245 }
246
247 if(screen == nullptr)
248 screen = qApp->primaryScreen();
249
250 const QRect avail = screen->availableGeometry();
251
252 // Horizontal clamp.
253 if(widPos.x() + widFullSize.width() > avail.x() + avail.width())
254 widPos.setX(avail.x() + avail.width() - widFullSize.width());
255 if(widPos.x() < avail.x())
256 widPos.setX(avail.x());
257
258 // Vertical clamp.
259 if(widPos.y() + widFullSize.height() > avail.y() + avail.height())
260 widPos.setY(avail.y() + avail.height() - widFullSize.height());
261 if(widPos.y() < avail.y())
262 widPos.setY(avail.y());
263
264 widget_->move(widPos);
265}
