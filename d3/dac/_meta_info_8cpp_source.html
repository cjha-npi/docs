57 /// \brief Resolve base path using Known Folder ID or environment variable fallback.
58 /// \details This functions attempts to resolve the base path for the given
59 /// folder ID by first querying the Windows API and falling back to an
60 /// environment variable if necessary.
61 /// \param folderId The Known Folder ID to query via SHGetKnownFolderPath.
62 /// \param envVar The environment variable to check if the API call fails.
63 /// \param envSuffix The suffix to append to the environment variable path.
64 /// \param pathNameForErrors The name of the path for error messages.
65 /// \warning On unrecoverable failure (cannot locate or validate the directory),
66 /// a native dialog is shown and the process is terminated (no return).
67 QString basePath(const KNOWNFOLDERID& folderId, const char* envVar, const QString& envSuffix, const QString& pathNameForErrors) noexcept
68 {
69 // String to hold the resolved base path.
70 QString base;
71
72 // Windows API attempt.
73 PWSTR p = nullptr;
74 const HRESULT hr = SHGetKnownFolderPath(folderId, KF_FLAG_DEFAULT, nullptr, &p);
75 if(SUCCEEDED(hr) && p) {
76 base = QString::fromWCharArray(p);
77 CoTaskMemFree(p);
78 }
79
80 // Environment variable fallback.
81 if(base.isEmpty()) {
82 if(const QString envPath = qEnvironmentVariable(envVar); !envPath.isEmpty()) {
83 base = envSuffix.isEmpty() ? envPath : QDir(envPath).filePath(envSuffix);
84 }
85 }
86
87 // Abort on failure to locate.
88 if(base.isEmpty()) {
89 npi::app::winStopMsgExe(QStringLiteral("Fatal Error: Could not locate the system's %1 directory.\n"
90 "This folder is required for application settings.").arg(pathNameForErrors));
91 }
92
93 // Abort if the path is missing or not a directory.
94 if(const QFileInfo info(base); !info.exists() || !info.isDir()) {
95 npi::app::winStopMsgExe(QStringLiteral("Fatal Error: The system's %1 path '%2' is missing or is not a "
96 "directory. Please ensure this folder exists.").arg(pathNameForErrors, QDir::fromNativeSeparators(base)));
97 }
98
99 return base;
100 }

102 /// \brief Create subdirectory under base path.
103 /// \details This function attempts to create a subdirectory
104 /// under the specified base path. On failure, a fatal error
105 /// message is displayed and the process is terminated.
106 /// \param base The base directory under which to create the subdirectory.
107 /// \param dirName The name of the subdirectory to create.
108 /// \warning On unrecoverable failure to create the directory,
109 /// a native dialog is shown and the process is terminated (no return).
110 QString createSubDir(const QString& base, const QString& dirName) noexcept
111 {
112 QDir root{base};
113 const QString target = root.filePath(dirName);
114 if(!root.mkpath(dirName)) {
115 npi::app::winStopMsgExe(QStringLiteral("Fatal Error: The required directory could not be created:\n%1\n\n"
116 "This may be due to a permissions issue or because a file with the same name already exists.")
117 .arg(QDir::fromNativeSeparators(target)));
118 }
119 return QDir::cleanPath(target);
120 }

161 /// \brief Schedules the initialization of application meta paths.
162 /// \details This function is designed to be called by `Q_CONSTRUCTOR_FUNCTION`
163 /// at application startup, which runs *before* `main()` and the
164 /// `QApplication` instance exist.
165 ///
166 /// To safely initialize paths that may depend on a `QApplication` context
167 /// this function uses `QTimer::singleShot(0, ...)` to post an asynchronous
168 /// task to the Qt event queue.
169 ///
170 /// The task, executed within a lambda, calls `npi::app::appProgramDataDir()`
171 /// and `npi::app::appPublicDocumentsDir()`. This task will only
172 /// execute *after* the event loop is started (i.e., when `app.exec()` runs)
173 /// **OR** `QCoreApplication::processEvents()` after `QApplication` is created in
174 /// the `main()` function.
175 ///
176 /// \note To force this initialization to run immediately after `QApplication`
177 /// is created (but before other code, like `MainWindow` construction),
178 /// call `QCoreApplication::processEvents()` right after the `QApplication`
179 /// object is instantiated in `main()`.
180 void scheduleInit()
181 {
182 QTimer::singleShot(0, []() {
183 npi::app::appProgramDataDir();
184 npi::app::appPublicDocumentsDir();
185 });
186 }
187
188 Q_COREAPP_STARTUP_FUNCTION(scheduleInit)
189}
