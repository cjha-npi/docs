1/**
2* \file StageWindow.h
3* \brief Independent utility for showing a top-level widget either instantly or with a source-origin animation.
4*
5* \details
6* This header defines `npi::ind::StageWindow`, a helper class for top-level `QWidget` display:
7* - `quick_()` positions and shows immediately.
8* - `animate_()` creates a temporary frameless overlay window that grows/moves from a source
9* rectangle (cursor or a "caller" widget) to the target widget's frame geometry, while the
10* real widget fades from 0.0 to 1.0 opacity at the end.
11*
12* Positioning guarantees:
13* - The target widget is adjusted to remain inside the available geometry of the chosen screen.
14* - Optional centering over a "center" widget.
15*
16* Safety and lifetime notes:
17* - `animate_()` does not capture `this`; it captures a `QPointer<QWidget>` to avoid use-after-free.
18* - If the target widget is deleted during the animation, the temporary overlay is still cleaned up.
19*
20* \warning Call these APIs from the GUI thread. Showing/moving widgets from worker threads is undefined.
21*/

31 /**
32 * \class StageWindow
33 * \brief Helper to display a top-level widget with optional animation and robust on-screen positioning.
34 * \details This class provides a convenient way to show a top-level widget with various display options,
35 * including immediate display and animated transitions. It handles positioning the widget within the
36 * available screen space and offers customization options for the animation duration and easing curve.
37 */

96 /// \brief Show immediately without animation.
97 /// \details
98 /// - Validates that `widget_` exists and is a top-level window.
99 /// - Positions the widget within current screen bounds.
100 /// - Uses `callWidgetCorrectShowMethod_()` to handle minimized/visible states robustly.
101 void quick_() const;

103 /// \brief Show with an animated transition from a small source rect (cursor or caller) to the target.
104 /// \details
105 /// Steps:
106 /// 1. Compute `begRect` from `caller_` or a tiny rect at `QCursor::pos()`.
107 /// 2. Call `positionWidget_()` to place within screen bounds and center if requested.
108 /// 3. Set target widget opacity to 0.0 and show it using `callWidgetCorrectShowMethod_()`.
109 /// 4. Create a temporary frameless tool window (overlay) at `begRect` with a background color derived
110 /// from the target widget palette (slightly lightened/darkened for contrast).
111 /// 5. Animate the overlay geometry from `begRect` to the target frame geometry.
112 /// 6. On finish, set target opacity to 1.0 if still alive and delete the overlay.
113 /// \note
114 /// - The overlay is `Qt::Tool` and `WA_TransparentForMouseEvents` to avoid focus/taskbar issues.
115 /// - The animation object uses `DeleteWhenStopped` for cleanup.
116 /// - The lambda does not capture `this` to avoid lifetime hazards.
117 void animate_() const;

131 /// \brief Compute and apply an on-screen position for `widget_`, respecting frames/title bars.
132 /// \details
133 /// Logic:
134 /// - Estimate full framed size if not yet shown (geometry == frameGeometry).
135 /// - If `center_` is set, center over it (using frame geometry when it is a window).
136 /// - Else, if the widget has never been displayed and sits at (0,0), center on the screen under the cursor.
137 /// - Select screen using the frame-geometry center of `widget_` (or `center_`); fallback to primary.
138 /// - Clamp position to available geometry (no overflow left/top/right/bottom).
139 void positionWidget_() const;
140
141 /// \brief Show the widget in a state-correct way.
142 /// \details
143 /// - If minimized: call `showNormal()`.
144 /// - Else: if visible, `hide()` first to reset painting, then `show()`.
145 /// - Else: `show()`.
