// Purpose:
// Provides taskbar-aware layout values to a Rainmeter skin.
//
// This plugin reads the Windows taskbar position and calculates the free
// horizontal taskbar area between:
// 1. the right edge of the last taskbar icon/button
// 2. the left edge of the system tray / notification area
//
// The intended use case is a Rainmeter skin that displays CPU/RAM histograms
// or bar graphs inside the free space on a horizontal Windows taskbar.
//
// Returned values:
// BegX = right edge of last taskbar icon/button before the tray
// EndX = left edge of system tray / notification area
// BegY = taskbar top Y coordinate
// EndY = taskbar bottom Y coordinate
// Disp = 0 if taskbar is left/right, 1 if taskbar is top/bottom
//
// Rainmeter usage example:
// [MeasureBegX]
// Measure=Plugin
// Plugin=CJ_SysGraphs
// Value=BegX
//
// [MeasureEndX]
// Measure=Plugin
// Plugin=CJ_SysGraphs
// Value=EndX
//
// [MeasureBegY]
// Measure=Plugin
// Plugin=CJ_SysGraphs
// Value=BegY
//
// [MeasureEndY]
// Measure=Plugin
// Plugin=CJ_SysGraphs
// Value=EndY
//
// [MeasureDisp]
// Measure=Plugin
// Plugin=CJ_SysGraphs
// Value=Disp
//
// Build and deploy:
// - Open x64 Native Tools Command Prompt for VS 2022
// - Navigate to the folder containing CJ_SysGraphs.cpp file,
// - Typing the destination e.g. 'D:\SomeFolder' when the command prompt has 'C:'
// as its origin will not work. To change drives first go to the drive by typing
// just the drive letter and colon e.g. 'D:' then navigate to the folder otherwise
// use command like 'cd /d D:\SomeFolder'.
// - If path contains spaces use double-quotes e.g. 'cd /d "D:\My Folder\Sub Folder"'
// - Run:
// cl /std:c++20 /EHsc /O2 /LD CJ_SysGraphs.cpp /link User32.lib Ole32.lib UIAutomationCore.lib /out:CJ_SysGraphs.dll
// - This will generate CJ_SysGraphs.dll along with other files extensions .obj, .exp, .lib.
// Delete all other files except CJ_SysGraphs.dll.
// - Deploy:
// - Copy CJ_SysGraphs.dll to C:\Users\UserName\AppData\Roaming\Rainmeter\Plugins (can also
// be found as %APPDATA%\Rainmeter\Plugins\‍). If the 'Plugins' folder does not exist at
// C:\Users\UserName\AppData\Roaming\Rainmeter then create it.
//
// Notes:
// - The DLL bitness must match Rainmeter bitness.
// - For normal modern Rainmeter installations, build the x64 DLL.
// - The plugin uses Win32 APIs for taskbar window rectangles.
// - The plugin uses UI Automation to find taskbar button rectangles.
// - The result is cached briefly so multiple Rainmeter measures in the same
// update cycle do not each trigger a full taskbar scan.
//
// =============================================================================
// Defines the NOMINMAX macro to remove the default min/max
// defined by Windows and to use the std::min/std::max instead.
// If this is not defined then using std::min/std::max causes
// serious errors.
#ifndef NOMINMAX
#define NOMINMAX
#endif
#include <Windows.h>
#include <UIAutomation.h>
#include <algorithm>
#include <cwchar>
#include <mutex>
// Link the required Windows libraries when building with MSVC.
// These pragmas are optional if the same libraries are passed on the command
// line after /link.
#pragma comment(lib, "User32.lib")
#pragma comment(lib, "Ole32.lib")
#pragma comment(lib, "UIAutomationCore.lib")
namespace {
// Identifies which value a particular Rainmeter measure requested.
// Each Rainmeter measure can load the same plugin DLL but request a
// different value using:
// Value=BegX
// Value=EndX
// Value=BegY
// Value=EndY
// Value=Disp
//
// The parsed result is stored per measure in the Measure structure.
enum class ValueKind {
BegX,
EndX,
BegY,
EndY,
Disp
};
// Stores all taskbar values calculated during one taskbar scan.
// begX: Right edge of the last taskbar icon/button before the tray.
// endX: Left edge of the system tray / notification area.
// begY: Top Y coordinate of the taskbar rectangle.
// endY: Bottom Y coordinate of the taskbar rectangle.
// disp: 0 (horizontal left/right)/ 1 (vertical left/right) taskbar orientation.
struct TaskbarVals {
int begX{};
int endX{};
int begY{};
int endY{};
int disp{};
};
// Per-measure data owned by Rainmeter through this plugin.
// Rainmeter creates one plugin instance per measure. For example,
// MeasureBegX and MeasureEndX each get their own Measure object.
// rm: Rainmeter measure handle. This is used by Rainmeter API functions.
// valueKind: The specific value this measure should return during Update().
struct Measure {
void* rm{};
ValueKind valueKind{ ValueKind::BegX };
};
// Global cached taskbar values.
// The skin may define five separate measures for BegX, EndX, BegY, EndY,
// and Disp. Without caching, all five measures could trigger five separate
// UI Automation scans during the same Rainmeter update cycle.
// This cache keeps the last calculated result for a short time.
struct CachedTaskbarVals {
TaskbarVals vals{};
ULONGLONG tick{};
bool valid{};
};
// Signature of Rainmeter's RmReadString function.
// The plugin gets this function dynamically from Rainmeter.dll instead of
// depending on a Rainmeter SDK header.
using RmReadStringFunc = LPCWSTR(__stdcall*)(void*, LPCWSTR, LPCWSTR, BOOL);
// Protects access to the global cache.
// Rainmeter usually calls measures in a controlled way, but the mutex keeps
// this safe if calls happen from different threads.
std::mutex cacheMutex;
// Stores the most recently calculated taskbar values.
CachedTaskbarVals cachedVals;
// Cache duration in milliseconds.
// 100 ms is short enough that taskbar changes are picked up quickly, but
// long enough to avoid repeated scans for multiple measures in the same
// update cycle.
constexpr ULONGLONG cacheMs = 100;
// Gets Rainmeter's RmReadString function from Rainmeter.dll.
// This allows the plugin to read options from the skin measure section,
// such as:
// Value=BegX
// If Rainmeter.dll or RmReadString cannot be found, the plugin falls back
// to default values.
RmReadStringFunc GetRmReadString() {
HMODULE rainmeter = GetModuleHandleW(L"Rainmeter.dll");
if (rainmeter == nullptr) {
return nullptr;
}
return reinterpret_cast<RmReadStringFunc>(GetProcAddress(rainmeter, "RmReadString"));
}
// Converts the Rainmeter Value= option text into a ValueKind.
// Unknown values fall back to BegX. This keeps the plugin predictable even
// if the skin contains a typo.
ValueKind ParseValueKind(LPCWSTR text) {
if (text == nullptr) {
return ValueKind::BegX;
}
if (_wcsicmp(text, L"BegX") == 0) {
return ValueKind::BegX;
}
if (_wcsicmp(text, L"EndX") == 0) {
return ValueKind::EndX;
}
if (_wcsicmp(text, L"BegY") == 0) {
return ValueKind::BegY;
}
if (_wcsicmp(text, L"EndY") == 0) {
return ValueKind::EndY;
}
if (_wcsicmp(text, L"Disp") == 0) {
return ValueKind::Disp;
}
return ValueKind::BegX;
}
// Safe wrapper around GetWindowRect.
// Returns false if the HWND or RECT pointer is invalid, or if the Win32 API
// call fails.
bool GetWindowRectSafe(HWND hwnd, RECT* rect) {
if (hwnd == nullptr || rect == nullptr) {
return false;
}
return GetWindowRect(hwnd, rect) != FALSE;
}
// Determines whether the taskbar is horizontal.
// Horizontal taskbar: Usually positioned at the top or bottom of the monitor.
// Vertical taskbar: Usually positioned at the left or right of the monitor.
// Main rule: If width >= height, treat it as horizontal.
// Fallback rule: If the taskbar covers most of the monitor width and less
// than half of the monitor height, also treat it as horizontal.
// The fallback is intentionally practical rather than overly complex.
bool IsHorizontalTaskbar(HWND taskbar, const RECT& taskbarRect) {
const int width = taskbarRect.right - taskbarRect.left;
const int height = taskbarRect.bottom - taskbarRect.top;
if (width >= height) {
return true;
}
HMONITOR monitor = MonitorFromWindow(taskbar, MONITOR_DEFAULTTONEAREST);
MONITORINFO monitorInfo{};
monitorInfo.cbSize = sizeof(monitorInfo);
if (monitor != nullptr && GetMonitorInfoW(monitor, &monitorInfo) != FALSE) {
const int monitorWidth = monitorInfo.rcMonitor.right - monitorInfo.rcMonitor.left;
const int monitorHeight = monitorInfo.rcMonitor.bottom - monitorInfo.rcMonitor.top;
// Practical fallback: a horizontal taskbar should cover most of the
// monitor width while using only a small part of the monitor height.
if (width > (monitorWidth / 2) && height < (monitorHeight / 2)) {
return true;
}
}
return false;
}
// Finds the right edge of the last taskbar button before the system tray.
// - taskbar: HWND of the main taskbar window.
// - startRight: Right edge of the Start button. This is used as the left boundary of
// the taskbar icon/button area.
// - trayLeft: Left edge of the tray/notification area. This is used as the right
// boundary of the taskbar icon/button area.
//
// Return: The largest right coordinate of any UI Automation button found between
// startRight and trayLeft.
//
// If UI Automation fails, the function returns startRight as a safe
// fallback. That gives the Rainmeter skin a conservative free area.
int GetLastTaskbarButtonRight(HWND taskbar, int startRight, int trayLeft) {
// UI Automation uses COM, so COM must be initialized on the calling
// thread before creating the automation object.
HRESULT coHr = CoInitializeEx(nullptr, COINIT_APARTMENTTHREADED);
// CoUninitialize must be called only when CoInitializeEx succeeded.
// SUCCEEDED covers S_OK and S_FALSE.
const bool shouldUninitialize = SUCCEEDED(coHr);
IUIAutomation* automation = nullptr;
HRESULT hr = CoCreateInstance(
CLSID_CUIAutomation,
nullptr,
CLSCTX_INPROC_SERVER,
IID_IUIAutomation,
reinterpret_cast<void**>(&automation));
if (FAILED(hr) || automation == nullptr) {
if (shouldUninitialize) {
CoUninitialize();
}
return startRight;
}
// Convert the taskbar HWND into a UI Automation element so the taskbar
// child controls can be inspected.
IUIAutomationElement* root = nullptr;
hr = automation->ElementFromHandle(taskbar, &root);
if (FAILED(hr) || root == nullptr) {
automation->Release();
if (shouldUninitialize) {
CoUninitialize();
}
return startRight;
}
// Build a condition that matches only UI Automation "Button" controls.
// Taskbar icons/buttons are exposed as buttons by UI Automation on
// normal Windows taskbars.
VARIANT variant{};
variant.vt = VT_I4;
variant.lVal = UIA_ButtonControlTypeId;
IUIAutomationCondition* condition = nullptr;
hr = automation->CreatePropertyCondition(UIA_ControlTypePropertyId, variant, &condition);
if (FAILED(hr) || condition == nullptr) {
root->Release();
automation->Release();
if (shouldUninitialize) {
CoUninitialize();
}
return startRight;
}
// Find all descendant button controls under the taskbar element.
// This can include the Start button, pinned/running app buttons, tray
// related controls, and other taskbar buttons. The rectangle filter
// below keeps only buttons inside the taskbar icon area.
IUIAutomationElementArray* array = nullptr;
hr = root->FindAll(TreeScope_Descendants, condition, &array);
if (FAILED(hr) || array == nullptr) {
condition->Release();
root->Release();
automation->Release();
if (shouldUninitialize) {
CoUninitialize();
}
return startRight;
}
int length = 0;
array->get_Length(&length);
int lastRight = startRight;
for (int ii = 0; ii < length; ++ii) {
IUIAutomationElement* element = nullptr;
if (FAILED(array->GetElement(ii, &element)) || element == nullptr) {
continue;
}
RECT rect{};
if (SUCCEEDED(element->get_CurrentBoundingRectangle(&rect))) {
const bool isValidRect = rect.right > rect.left && rect.bottom > rect.top;
// Keep only buttons between the Start button and the system
// tray. This removes Start, tray, and unrelated taskbar buttons
// from the result.
const bool isInsideTaskArea = rect.left >= startRight && rect.right <= trayLeft;
if (isValidRect && isInsideTaskArea) {
lastRight = std::max(lastRight, static_cast<int>(rect.right));
}
}
element->Release();
}
array->Release();
condition->Release();
root->Release();
automation->Release();
if (shouldUninitialize) {
CoUninitialize();
}
return lastRight;
}
// Reads the current taskbar values.
// This is the main calculation function used by the plugin. It:
// 1. finds the main taskbar window,
// 2. reads the taskbar rectangle,
// 3. determines whether the taskbar is horizontal,
// 4. finds the Start button and tray rectangles,
// 5. scans taskbar buttons through UI Automation,
// 6. returns all values needed by the Rainmeter skin.
TaskbarVals ReadTaskbarVals() {
TaskbarVals vals{};
HWND taskbar = FindWindowW(L"Shell_TrayWnd", nullptr);
if (taskbar == nullptr) {
return vals;
}
RECT taskbarRect{};
if (!GetWindowRectSafe(taskbar, &taskbarRect)) {
return vals;
}
vals.begY = static_cast<int>(taskbarRect.top);
vals.endY = static_cast<int>(taskbarRect.bottom);
const bool isHorizontal = IsHorizontalTaskbar(taskbar, taskbarRect);
vals.disp = isHorizontal ? 1 : 0;
// If the taskbar is vertical, the skin should not be displayed in the
// taskbar free area. X values are returned as 0 because they are not
// useful for the horizontal layout calculation.
if (!isHorizontal) {
vals.begX = 0;
vals.endX = 0;
return vals;
}
// Get the Start button and system tray windows.
// - Start: Used to determine the left boundary of the taskbar icon area.
// - TrayNotifyWnd: Used to determine the right boundary of the taskbar icon area.
HWND start = FindWindowExW(taskbar, nullptr, L"Start", nullptr);
HWND tray = FindWindowExW(taskbar, nullptr, L"TrayNotifyWnd", nullptr);
RECT startRect{};
RECT trayRect{};
const bool hasStart = GetWindowRectSafe(start, &startRect);
const bool hasTray = GetWindowRectSafe(tray, &trayRect);
// Fallbacks:
// If Start is not found, use the taskbar left edge.
// If tray is not found, use the taskbar right edge.
// These fallbacks keep the plugin usable even if Windows changes some
// internal taskbar window class names.
const int startRight = hasStart ? static_cast<int>(startRect.right) : static_cast<int>(taskbarRect.left);
const int trayLeft = hasTray ? static_cast<int>(trayRect.left) : static_cast<int>(taskbarRect.right);
vals.begX = GetLastTaskbarButtonRight(taskbar, startRight, trayLeft);
vals.endX = trayLeft;
return vals;
}
// Returns cached taskbar values when the cache is still fresh.
// This prevents repeated UI Automation scans when multiple Rainmeter
// measures update at nearly the same time.
TaskbarVals GetCachedTaskbarVals() {
std::lock_guard<std::mutex> lock(cacheMutex);
const ULONGLONG now = GetTickCount64();
if (cachedVals.valid && now >= cachedVals.tick && (now - cachedVals.tick) < cacheMs) {
return cachedVals.vals;
}
cachedVals.vals = ReadTaskbarVals();
cachedVals.tick = now;
cachedVals.valid = true;
return cachedVals.vals;
}
// Returns the specific value requested by the current Rainmeter measure.
// The taskbar scan calculates all values at once. This function selects the
// one value that belongs to this measure's Value= option.
double GetRequestedValue(const Measure* measure, const TaskbarVals& vals) {
if (measure == nullptr) {
return 0.0;
}
switch (measure->valueKind) {
case ValueKind::BegX: return static_cast<double>(vals.begX);
case ValueKind::EndX: return static_cast<double>(vals.endX);
case ValueKind::BegY: return static_cast<double>(vals.begY);
case ValueKind::EndY: return static_cast<double>(vals.endY);
case ValueKind::Disp: return static_cast<double>(vals.disp);
default: return 0.0;
}
}
}
// Called by Rainmeter when a measure instance using this plugin is created.
// - data: Output pointer where the plugin stores per-measure data.
// - rm: Rainmeter measure handle for this measure instance.
extern "C" __declspec(dllexport) void Initialize(void** data, void* rm) {
Measure* measure = new Measure;
measure->rm = rm;
*data = measure;
}
// Called by Rainmeter when the measure is loaded or refreshed.
// This reads the Value= option from the Rainmeter skin and stores which value
// this measure should return during Update().
extern "C" __declspec(dllexport) void Reload(void* data, void* rm, double* maxValue) {
Measure* measure = static_cast<Measure*>(data);
if (measure == nullptr) {
return;
}
measure->rm = rm;
RmReadStringFunc rmReadString = GetRmReadString();
LPCWSTR valueText = L"BegX";
if (rmReadString != nullptr) {
valueText = rmReadString(rm, L"Value", L"BegX", TRUE);
}
measure->valueKind = ParseValueKind(valueText);
// This plugin returns direct numeric values. It does not need Rainmeter to
// use a fixed maximum value.
if (maxValue != nullptr) {
*maxValue = 0.0;
}
}
// Called by Rainmeter whenever the measure updates.
// This gets the current cached taskbar values and returns only the value
// requested by this specific measure.
extern "C" __declspec(dllexport) double Update(void* data) {
Measure* measure = static_cast<Measure*>(data);
const TaskbarVals vals = GetCachedTaskbarVals();
return GetRequestedValue(measure, vals);
}
// Called by Rainmeter when the measure instance is destroyed.
// This releases the per-measure data allocated in Initialize().
extern "C" __declspec(dllexport) void Finalize(void* data) {
Measure* measure = static_cast<Measure*>(data);
delete measure;
}
