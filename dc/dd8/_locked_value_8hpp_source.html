23 /**
24 * \brief Stores a value protected by a `QReadWriteLock`.
25 * \details
26 * `LockedValue` supports three access styles:
27 *
28 * - Copy-based operations through `read_()` and `write_()`.
29 * - Function-based operations through `readWith_()` and `writeWith_()`.
30 * - Scoped reference access through `readAccess_()` and `writeAccess_()`.
31 *
32 * Snapshot operations hold the lock only while copying or assigning the value.
33 * Scoped access objects retain the corresponding lock until the access object is
34 * destroyed.
35 *
36 * Individual members can be read or written through pointers to data members:
37 * \code
38 * const QString directory = config.read_(&Config::directory_);
39 * config.write_(&Config::maxFolderMb_, 512u);
40 * \endcode
41 *
42 * Multiple related operations can be performed under one lock:
43 * \code
44 * auto config = lockedConfig.writeAccess_();
45 * config->directory_ = newDirectory;
46 * config->maxFolderMb_ = 512u;
47 * \endcode
48 *
49 * \tparam T Non-const, non-volatile, non-array object type to protect.
50 *
51 * \warning The object must outlive all `ReadAccess` and `WriteAccess` objects
52 * created from it.
53 * \warning Do not call another locking function on the same `LockedValue` while
54 * already holding one of its scoped access objects.
55 */
56 template <typename T>
57 class LockedValue final {
58 static_assert(std::is_object_v<T>,
59 "LockedValue<T> requires T to be an object type.");
60
61 static_assert(!std::is_array_v<T>,
62 "LockedValue<T> does not support array types.");
63
64 static_assert(!std::is_const_v<T> && !std::is_volatile_v<T>,
65 "LockedValue<T> requires a non-const, non-volatile value type.");
66
67 public:
68 /// For outside code that needs to discover the stored type.
69 using ValueType = T;
70
71 /**
72 * \brief Provides scoped read-only access to the protected value.
73 * \details
74 * Acquires the internal lock for reading during construction and releases it
75 * when destroyed. The access object cannot be copied or moved.
76 *
77 * \warning
78 * References and pointers obtained through this object must not be retained
79 * after the access object is destroyed.
80 */
81 class ReadAccess final {
82 public:
83 /// Disables copying because each access object owns one read lock.
84 ReadAccess(const ReadAccess&) = delete;
85
86 /// Disables copy assignment because lock ownership cannot be copied.
87 ReadAccess& operator=(const ReadAccess&) = delete;
88
89 /// Disables moving because `QReadLocker` does not transfer ownership.
90 ReadAccess(ReadAccess&&) = delete;
91
92 /// Disables move assignment because lock ownership cannot be transferred.
93 ReadAccess& operator=(ReadAccess&&) = delete;
94
95 /// Releases the read lock.
96 ~ReadAccess() = default;
97
98 /// \brief Provides pointer-style read-only access to the protected value.
99 /// \return Pointer to the protected value.
100 /// \warning The returned pointer must not outlive this access object.
101 [[nodiscard]] const T* operator->() const noexcept
102 {
103 return &value_;
104 }
105
106 /// \brief Provides reference-style read-only access to the protected value.
107 /// \return Constant reference to the protected value.
108 /// \warning The returned reference must not outlive this access object.
109 [[nodiscard]] const T& operator*() const noexcept
110 {
111 return value_;
112 }
113
114 private:
115 friend class LockedValue;
116
117 /// \brief Acquires read access to the supplied value.
118 /// \param lock Lock protecting the value.
119 /// \param value Protected value.
120 ReadAccess(QReadWriteLock& lock, const T& value)
121 : locker_{&lock}, value_{value} {}
122
123 QReadLocker locker_; ///< Owns the read lock for this access object's lifetime.
124 const T& value_; ///< Protected value exposed as read-only.
125 };

246 /// \brief Returns a snapshot copy of the complete protected value.
247 /// \return Copy of the protected value.
248 /// \note The read lock is released before this function returns.
249 /// \note This function is available only when `T` is copy-constructible.
250 [[nodiscard]] T read_() const
251 requires std::copy_constructible<T>
252 {
253 QReadLocker locker{&lock_};
254 return value_;
255 }

257 /// \brief Returns a snapshot copy of one data member.
258 /// \tparam M Member value type.
259 /// \param member Pointer to the data member to read.
260 /// \return Copy of the selected member.
261 ///
262 /// Only the selected member is copied. The complete protected value is not
263 /// copied.
264 ///
265 /// \pre `member` must not be null.
266 template <typename M>
267 [[nodiscard]] std::remove_cv_t<M> read_(M T::* member) const
268 requires (!std::is_array_v<M> &&
269 std::copy_constructible<std::remove_cv_t<M>>)
270 {
271 QReadLocker locker{&lock_};
272 return value_.*member;
273 }

275 /// \brief Replaces the complete protected value.
276 /// \tparam U Source value type.
277 /// \param value Value assigned to the protected object.
278 ///
279 /// The source value is perfectly forwarded to the assignment operator of `T`.
280 template <typename U>
281 void write_(U&& value)
282 requires std::is_assignable_v<T&, U&&>
283 {
284 QWriteLocker locker{&lock_};
285 value_ = std::forward<U>(value);
286 }
287
288 /// \brief Assigns a new value to one data member.
289 /// \tparam M Member value type.
290 /// \tparam U Source value type.
291 /// \param member Pointer to the data member to modify.
292 /// \param value Value assigned to the selected member.
293 ///
294 /// \pre `member` must not be null.
295 template <typename M, typename U>
296 void write_(M T::* member, U&& value)
297 requires std::is_assignable_v<M&, U&&>
298 {
299 QWriteLocker locker{&lock_};
300 value_.*member = std::forward<U>(value);
301 }

303 /// \brief Invokes a callable with read-only access to the protected value.
304 /// \details
305 /// Holds the read lock for the complete callable invocation. The callable receives
306 /// the protected value as `const T&`.
307 /// \tparam F Callable type accepting `const T&`.
308 /// \param function Callable to invoke while holding the read lock.
309 /// \return The callable's return value, or nothing when it returns `void`.
310 /// \warning The callable must not return a pointer or reference to protected data
311 /// because the read lock is released when this function returns.
312 template <typename F>
313 decltype(auto) readWith_(F&& function) const
314 {
315 QReadLocker locker{&lock_};
316 return std::invoke(
317 std::forward<F>(function),
318 std::as_const(value_));
319 }
320
321 /// \brief Invokes a callable with mutable access to the protected value.
322 /// \details
323 /// Holds the write lock for the complete callable invocation. The callable receives
324 /// the protected value as `T&`.
325 /// \tparam F Callable type accepting `T&`.
326 /// \param function Callable to invoke while holding the write lock.
327 /// \return The callable's return value, or nothing when it returns `void`.
328 /// \warning The callable must not return a pointer or reference to protected data
329 /// because the write lock is released when this function returns.
330 template <typename F>
331 decltype(auto) writeWith_(F&& function)
332 {
333 QWriteLocker locker{&lock_};
334 return std::invoke(
335 std::forward<F>(function),
336 value_);
337 }
338
339 /// \brief Acquires scoped read-only access to the complete protected value.
340 /// \return Access object that retains the read lock until destroyed.
341 ///
342 /// Use this function when several members must be read consistently without
343 /// copying the complete value.
344 [[nodiscard]] ReadAccess readAccess_() const
345 {
346 return ReadAccess{lock_, value_};
347 }
348
349 /// \brief Acquires scoped mutable access to the complete protected value.
350 /// \return Access object that retains the write lock until destroyed.
351 ///
352 /// Use this function when several related modifications must be performed as
353 /// one synchronized operation.
354 [[nodiscard]] WriteAccess writeAccess_()
355 {
356 return WriteAccess{lock_, value_};
357 }
358
359 private:
360 mutable QReadWriteLock lock_; ///< Synchronizes access to `value_`.
361 T value_; ///< Value protected by `lock_`.
362 };
