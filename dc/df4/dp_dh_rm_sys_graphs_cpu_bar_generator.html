-- Description
-- ============================================================================
--
-- Generates the Rainmeter include file that contains one CPU measure, one CPU
-- bar meter, and one CPU text meter for each logical CPU thread.
--
-- The generated file is included by the main Rainmeter skin using @Include.
-- Rainmeter reads @Include files only when the skin is loaded/refreshed, so this
-- script refreshes the skin only when the generated file actually changes.
--
-- Use it in the Rainmeter skin as:
--
-- [GenerateCpuBars]
-- Measure=Script
-- ScriptFile=#CURRENTPATH#CJ_SysGraphs_GenCpuBars.lua
-- OutputFile=#CURRENTPATH#CJ_SysGraphs_CpuBars.inc
-- UpdateDivider=-1
--
-- @Include=#CURRENTPATH#CJ_SysGraphs_CpuBars.inc
--
-- ============================================================================
-- Called automatically by Rainmeter when this script measure is initialized.
-- This usually happens when the skin is loaded or refreshed.
function Initialize()
-- Read the OutputFile option from the Rainmeter script measure.
-- Example: OutputFile=#CURRENTPATH#CJ_SysGraphs_CpuBars.inc
local output_file = SELF:GetOption("OutputFile")
-- Convert the Rainmeter path to an absolute Windows path.
-- This allows io.open() to write to the correct file location.
output_file = SKIN:MakePathAbsolute(output_file)
-- Read the number of logical processors reported by Windows.
-- This is usually CPU threads, not necessarily physical CPU cores.
-- If the environment variable is missing or invalid, fall back to 1.
local cpu_count = tonumber(os.getenv("NUMBER_OF_PROCESSORS")) or 1
-- Build the full text that should be written to the generated include file.
local text = BuildCpuBars(cpu_count)
-- Read the existing generated file.
-- If the file already contains exactly the same text, nothing needs to be
-- written and the skin does not need to be refreshed.
local old_text = ReadFile(output_file)
if old_text == text then
return
end
-- Write the new generated text.
-- If writing succeeds, refresh the skin so Rainmeter re-reads the @Include
-- file and loads the newly generated sections.
if WriteFile(output_file, text) then
SKIN:Bang("!Refresh")
end
end
-- Builds the Rainmeter code for all CPU bars.
-- cpu_count:
-- Number of logical CPU threads for which sections should be generated.
-- returns:
-- A single string containing all generated Rainmeter sections.
function BuildCpuBars(cpu_count)
-- Store all generated lines in a table first.
-- This is better than repeatedly concatenating one large string.
local lines = {}
-- Header added to the generated file for clarity.
-- This also helps identify whether the file was generated by this script.
table.insert(lines, "; ============================================================================")
table.insert(lines, "; File: CJ_SysGraphs_CpuBars.inc")
table.insert(lines, "; Generated/Edited automatically by CJ_SysGraphs_GenCpuBars.lua.")
table.insert(lines, "; --- DO NOT EDIT MANUALLY! ---")
table.insert(lines, "; ============================================================================")
table.insert(lines, "")
table.insert(lines, "; CPU Core Count: " .. cpu_count)
table.insert(lines, "")
-- Generate measures and meters per CPU thread in a loop.
for ii = 1, cpu_count do
-- Rainmeter CPU processor indexes start at 1 for the first logical CPU.
-- Meter X positions start at 0, so the first bar uses index 0, the second
-- bar uses index 1, and so on.
local one_less = ii - 1
local big_gaps = math.floor(one_less / 4)
local sml_gaps = one_less - big_gaps
table.insert(lines, "; ----------- Core " .. ii .. " -----------")
-- ------------------------------------------------------------------------
-- CPU core load measure
-- ------------------------------------------------------------------------
-- Creates one CPU load measure for the current core index.
table.insert(lines, "[MeasureCPU" .. ii .. "Load]")
table.insert(lines, "Measure=CPU")
table.insert(lines, "Processor=" .. ii)
table.insert(lines, "")
-- ------------------------------------------------------------------------
-- Averaged CPU core load measure
-- ------------------------------------------------------------------------
-- Creates one CPU load measure for the current core index.
-- AverageSize uses the effective tick window, so the matching tick meter
-- shows averaged load instead of the instantaneous load.
table.insert(lines, "[MeasureCPU" .. ii .. "Tick]")
table.insert(lines, "Measure=CPU")
table.insert(lines, "Processor=" .. ii)
table.insert(lines, "Group=TickWindowMeasures")
table.insert(lines, "AverageSize=[TickWindow]")
table.insert(lines, "DynamicVariables=1")
table.insert(lines, "")
-- ------------------------------------------------------------------------
-- CPU core text measures
-- ------------------------------------------------------------------------
-- Averages and rounds the current core load for the displayed text.
-- The text value updates on the work update interval, not every skin update.
table.insert(lines, "[MeasureCPU" .. ii .. "TextRaw]")
table.insert(lines, "Measure=CPU")
table.insert(lines, "Processor=" .. ii)
table.insert(lines, "AverageSize=[WorkUpdate]")
table.insert(lines, "DynamicVariables=1")
table.insert(lines, "")
table.insert(lines, "[MeasureCPU" .. ii .. "Text]")
table.insert(lines, "Measure=Calc")
table.insert(lines, "Formula=Round(MeasureCPU" .. ii .. "TextRaw)")
table.insert(lines, "MinValue=0")
table.insert(lines, "MaxValue=100")
table.insert(lines, "UpdateDivider=[WorkUpdate]")
table.insert(lines, "DynamicVariables=1")
table.insert(lines, "")
-- ------------------------------------------------------------------------
-- CPU core bar meter
-- ------------------------------------------------------------------------
-- Draws the live load bar for the current core.
-- The bar is positioned from the generated core layout values.
table.insert(lines, "[MeterCPU" .. ii .. "Bar]")
table.insert(lines, "Meter=Bar")
table.insert(lines, "Group=BarDependentMeter")
table.insert(lines, "AntiAlias=1")
table.insert(lines, "MeasureName=MeasureCPU" .. ii .. "Load")
table.insert(lines, "X=([BarSectionX] + (" .. one_less .. " * [BarW]) + (" .. big_gaps .. " * [BigGap]) + (" .. sml_gaps .. " * #Gap#))")
table.insert(lines, "Y=0")
table.insert(lines, "W=[BarW]")
table.insert(lines, "H=[SkinHeight]")
table.insert(lines, "BarColor=[##ThemeName#_CPUBar]")
table.insert(lines, "SolidColor=[##ThemeName#_CPUBarBase]")
table.insert(lines, "Hidden=(([BarW] = 0) ? 1 : 0)")
table.insert(lines, "DynamicVariables=1")
table.insert(lines, "")
-- ------------------------------------------------------------------------
-- CPU core index text meter
-- ------------------------------------------------------------------------
-- Draws the core index at the top of the current core bar.
-- Every fourth core remains eligible for display when individual core
-- labels are disabled, while the other labels follow ShowBarIndividualText.
-- All labels are hidden when bar details are disabled or no bar width exists.
table.insert(lines, "[MeterCPU" .. ii .. "Text]")
table.insert(lines, "Meter=String")
table.insert(lines, "Group=BarDetails | BarDependentMeter")
table.insert(lines, "Text=" .. ii)
table.insert(lines, "AntiAlias=1")
table.insert(lines, "X=(([BarSectionX] + (" .. one_less .. " * [BarW]) + (" .. big_gaps .. " * [BigGap]) + (" .. sml_gaps .. " * #Gap#)) + ([BarW] / 2))")
table.insert(lines, "Y=0")
table.insert(lines, "W=[BarW]")
table.insert(lines, "H=[BarTextH]")
table.insert(lines, "StringAlign=Center")
table.insert(lines, "FontFace=#FontFamilyName#")
table.insert(lines, "FontSize=#FontPointSize#")
table.insert(lines, "FontColor=[##ThemeName#_BarText]")
table.insert(lines, "SolidColor=[##ThemeName#_BarTextBase]")
if one_less % 4 == 0 then
table.insert(lines, "Hidden=((([BarW] = 0) || ((#ShowBarDetails# = 0) && (#ShowBarDetailsTemp# = 0))) ? 1 : 0)")
else
table.insert(lines, "Hidden=((([BarW] = 0) || ([ShowBarIndividualText] = 0) || ((#ShowBarDetails# = 0) && (#ShowBarDetailsTemp# = 0))) ? 1 : 0)")
end
table.insert(lines, "DynamicVariables=1")
table.insert(lines, "")
-- ------------------------------------------------------------------------
-- CPU core averaged-load tick meter
-- ------------------------------------------------------------------------
-- Draws a 2 px horizontal marker over the current core bar.
-- The marker position is based on the averaged CPU load from MeasureCPUxxTick.
-- HalfGap lets the marker extend slightly into the gaps between bars.
table.insert(lines, "[MeterCPU" .. ii .. "Tick]")
table.insert(lines, "Meter=Shape")
table.insert(lines, "Group=BarDependentMeter")
table.insert(lines, "X=([BarSectionX] + (" .. one_less .. " * [BarW]) + (" .. big_gaps .. " * [BigGap]) + (" .. sml_gaps .. " * #Gap#) - [HalfGap])")
table.insert(lines, "Y=(Max(0, Min(([SkinHeight] - 2), ([SkinHeight] - 2 - Round(([SkinHeight] - 2) * ([MeasureCPU" .. ii .. "Tick] / 100))))))")
table.insert(lines, "W=([BarW] + #Gap#)")
table.insert(lines, "H=2")
table.insert(lines, "SolidColor=[##ThemeName#_CPUCoreAvg]")
table.insert(lines, "Hidden=(([BarW] = 0) ? 1 : 0)")
table.insert(lines, "DynamicVariables=1")
table.insert(lines, "")
-- ------------------------------------------------------------------------
-- CPU core interaction meter
-- ------------------------------------------------------------------------
-- Adds the hover area for the current core label.
-- The meter is almost transparent but still receives mouse input.
-- Hover temporarily brings the skin above the taskbar, and scrolling changes
-- the tick-window value used by the averaged-load tick markers.
table.insert(lines, "[CPU" .. ii .. "Interaction]")
table.insert(lines, "Meter=Shape")
table.insert(lines, "Group=BarDependentMeter | TickWindowMeters")
table.insert(lines, "X=([BarSectionX] + (" .. one_less .. " * [BarW]) + (" .. big_gaps .. " * [BigGap]) + (" .. sml_gaps .. " * #Gap#))")
table.insert(lines, "Y=0")
table.insert(lines, "Shape=Rectangle 0,0,[BarW],[BarTextH] | Fill Color 0,0,0,1 | StrokeWidth 0")
table.insert(lines, "ToolTipTitle=CPU Core " .. ii .. ": [MeasureCPU" .. ii .. "Text]%")
table.insert(lines, "ToolTipText=[TickWindow_ToolTip]")
table.insert(lines, "Hidden=(([BarW] = 0) ? 1 : 0)")
table.insert(lines, "MouseOverAction=[!SetVariable CurZPos \"1\"][!ZPos 1]")
table.insert(lines, "MouseLeaveAction=[!SetVariable CurZPos \"2\"][!ZPos 2]")
table.insert(lines, "MouseScrollUpAction=[!EnableMeasure TickWindowMs_Inc][!UpdateMeasure TickWindowMs_Inc][!DisableMeasure TickWindowMs_Inc]")
table.insert(lines, "MouseScrollDownAction=[!EnableMeasure TickWindowMs_Dec][!UpdateMeasure TickWindowMs_Dec][!DisableMeasure TickWindowMs_Dec]")
table.insert(lines, "DynamicVariables=1")
table.insert(lines, "")
end
-- Convert the line table into one generated text block.
return table.concat(lines, "\n")
end
-- Reads a full file as text.
-- path:
-- Absolute path of the file to read.
-- returns:
-- The file contents, or an empty string if the file cannot be opened.
function ReadFile(path)
local file = io.open(path, "r")
if not file then
return ""
end
local text = file:read("*a")
file:close()
return text or ""
end
-- Writes text to a file.
-- path:
-- Absolute path of the file to write.
-- text:
-- Text content to write to the file.
-- returns:
-- true if writing succeeded, false otherwise.
function WriteFile(path, text)
local file = io.open(path, "w")
if not file then
print("Failed to write CPU bars include file: " .. path)
return false
end
file:write(text)
file:close()
return true
end
