32 /// \brief Converts Windows `DWORD` error code into format "<text> (Code <number>).".
33 /// \details
34 /// The function uses `FormatMessageW()` with the system message table to
35 /// retrieve the description associated with `DWORD` error code. Windows
36 /// allocates the message buffer internally, and the function releases
37 /// that buffer using `LocalFree()` before returning.
38 ///
39 /// Windows system messages commonly contain trailing carriage returns, line
40 /// feeds, spaces, or tabs. These trailing characters are removed so the
41 /// returned text can be embedded cleanly into log messages, dialogs, and
42 /// exception descriptions.
43 ///
44 /// \param errorCode
45 /// Windows system error code to convert. This is commonly the value captured
46 /// from `GetLastError()` immediately after a failed Windows API call.
47 ///
48 /// \return
49 /// Returns `std::wstring` human-readable system message without trailing
50 /// whitespace. If supplied code is `0` then it returns "No error (Code 0)."
51 /// If no system message is available or if `FormatMessageW()` fails, it
52 /// returns "Unknown error, FormatMessageW failed (Code <number>)."
53 ///
54 /// \note
55 /// Call `GetLastError()` before invoking this function and store its result
56 /// locally. Calling other Windows API functions first may overwrite the
57 /// thread's last-error value.
58 inline std::wstring winErrorText(DWORD errorCode)
59 {
60 if(errorCode == 0)
61 return L"No error (Code 0).";
62
63 constexpr DWORD flags =
64 FORMAT_MESSAGE_ALLOCATE_BUFFER |
65 FORMAT_MESSAGE_FROM_SYSTEM |
66 FORMAT_MESSAGE_IGNORE_INSERTS;
67
68 // Buffer to hold text
69 wchar_t* buf = nullptr;
70
71 // Ask Windows to allocate and fill the message buffer.
72 // The buffer must be released with LocalFree().
73 const DWORD len = ::FormatMessageW(
74 flags, nullptr, errorCode, 0, reinterpret_cast<LPWSTR>(&buf), 0, nullptr);
75
76 // Construct message from buffer
77 std::wstring msg;
78 if(len != 0 && buf != nullptr) {
79 msg.assign(buf, len);
80
81 // FormatMessageW commonly appends CR/LF and spaces.
82 // Remove them so the text is safe to embed in one-line diagnostics.
83 while(!msg.empty()
84 && (msg.back() == L'\r'
85 || msg.back() == L'\n'
86 || msg.back() == L' '
87 || msg.back() == L'\t')) {
88 msg.pop_back();
89 }
90 }
91 else {
92 // Formatting failed
93 msg = L"Unknown error, FormatMessageW failed";
94 }
95
96 // Append error code
97 msg.append(L" (Code ")
98 .append(std::to_wstring(errorCode))
99 .append(L").");
100
101 // buffer must be released
102 if(buf != nullptr)
103 ::LocalFree(buf);
104
105 return msg;
106 }
107
108 /// \brief Calls `GetLastError()` and returns error in format "<text> (Code <number>).".
109 /// \return
110 /// Returns `std::wstring` human-readable system message without trailing
111 /// whitespace. If `GetLastError()` is `0` then it returns "No error (Code 0)."
112 /// If no system message is available or if `FormatMessageW()` fails, it
113 /// returns "Unknown error, FormatMessageW failed (Code <number>)."
114 inline std::wstring winLastErrorText()
115 {
116 return winErrorText(::GetLastError());
117 }
