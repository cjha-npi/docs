1/**
2* \file WinMsgBoxExe.cpp
3* \brief Implements helper-process based Windows message boxes.
4*
5* \details
6* This file launches `npiMsgBox.exe` to show native Windows message boxes from
7* a separate process. This allows information and warning messages to be shown
8* without blocking the caller thread.
9*
10* The stop message variant launches the helper process and then terminates the
11* current process. This is intended for known fatal paths where the caller wants
12* to show a final message but does not want normal shutdown or CRT abort handling.
13*
14* The command line passed to `npiMsgBox.exe` has the form:
15* "full\path\to\npiMsgBox.exe" "Application Title" "Message" "Severity"
16*
17* Severity:
18* - 0 = information
19* - 1 = warning
20* - 2 = stop/error
21*
22* Message text is sanitized before being appended to the command line. Unsupported
23* characters are replaced, quotes are converted to apostrophes, and backslashes
24* are converted to forward slashes to avoid command-line parsing issues.
25*/

130 /// \brief Sanitizes and appends one title character to the command-line prefix.
131 bool cmdLineBufAppendTitleChar(unsigned int uChr) noexcept
132 {
133 if(cmdLinePos + 1u >= cmdLineMax) return false;
134 if(uChr < 32u || uChr > 126u) cmdLineBuf[cmdLinePos++] = L'?';
135 else if(uChr == '"') cmdLineBuf[cmdLinePos++] = L'\'';
136 else if(uChr == '\\') cmdLineBuf[cmdLinePos++] = L'/';
137 else cmdLineBuf[cmdLinePos++] = static_cast<wchar_t>(uChr);
138 cmdLineBuf[cmdLinePos] = L'\0';

142 /// \brief Appends the application title used as the message-box title to the command-line prefix.
143 bool cmdLineBufAppendTitle() noexcept
144 {
145 const char* data = npi::app::name.data();
146 const qsizetype size = npi::app::name.size();
147 for(qsizetype ii = 0; ii < size; ++ii) {
148 if(!cmdLineBufAppendTitleChar(static_cast<unsigned char>(data[ii])))
149 return false;
150 }
151 return true;

154 /// \brief Builds the helper executable path and reusable command-line prefix.
155 /// The command-line prefix buffer is filled with
156 /// `"full\path\to\npiMsgBox.exe" "Application Title" "`
157 /// This enables further messages to copy the prepared prefix and append
158 /// their message and severity. Info and warning messages use local buffers,
159 /// while fatal/stop messages use the dedicated static stop buffer.
160 /// \note This function is guarded because the prefix buffers are shared.
161 bool prepare() noexcept
162 {
163 while(true) {
164 if(prefixPrepared.load(std::memory_order_acquire))
165 return true;
166
167 bool expected = false;
168 if(prefixPreparing.compare_exchange_strong(expected, true, std::memory_order_acq_rel))
169 break;
170
171 prefixPreparing.wait(true, std::memory_order_acquire);
172 }
173
174 auto finish = [](bool success) noexcept {
175 if(success)
176 prefixPrepared.store(true, std::memory_order_release);
177 prefixPreparing.store(false, std::memory_order_release);
178 prefixPreparing.notify_all();
179 return success;
180 };
181
182 exePathPos = 0;
183 cmdLinePos = 0;
184
185 const DWORD n = ::GetModuleFileNameW(nullptr, appExeFullPath, exePathMax);
186 if(n == 0 || n >= exePathMax)
187 return finish(false);
188
189 // Find the directory of the current executable and place npiMsgBox.exe next to it.
190 size_t sep = 0;
191 for(size_t ii = 0; ii < n; ++ii) {
192 if(appExeFullPath[ii] == '\\' || appExeFullPath[ii] == '/')
193 sep = ii;
194 }
195
196 if(sep == 0 || sep == n) return finish(false);
197
198 if(!cmdLineBufAppend(L'\"')) return finish(false);
199 for(size_t ii = 0; ii < sep + 1; ++ii) {
200 if(!exePathBufAppend(appExeFullPath[ii]))
201 return finish(false);
202 if(!cmdLineBufAppend(appExeFullPath[ii]))
203 return finish(false);
204 }
205
206 for(size_t ii = 0; msgBoxExeName[ii] != L'\0'; ++ii) {
207 if(!exePathBufAppend(msgBoxExeName[ii]))
208 return finish(false);
209 if(!cmdLineBufAppend(msgBoxExeName[ii]))
210 return finish(false);
211 }
212
213 if(!cmdLineBufAppend(L'\"')) return finish(false);
214 if(!cmdLineBufAppend(L' ')) return finish(false);
215 if(!cmdLineBufAppend(L'\"')) return finish(false);
216 if(!cmdLineBufAppendTitle()) return finish(false);
217 if(!cmdLineBufAppend(L'\"')) return finish(false);
218 if(!cmdLineBufAppend(L' ')) return finish(false);
219 if(!cmdLineBufAppend(L'\"')) return finish(false);
220
221 return finish(true);
222 }

224 /// \brief Copies the prepared command-line prefix into a per-call buffer.
225 /// \details Info/warning messages use a local buffer so multiple callers do not modify shared state.
226 void copyCmdLineBuf(wchar_t* buf, size_t& pos) noexcept
227 {
228 pos = 0;
229 for(size_t ii = 0; ii < cmdLinePos; ++ii)
230 buf[pos++] = cmdLineBuf[ii];
231 buf[pos] = L'\0';
232 }

234 /// \brief Sanitizes and appends one message character.
235 /// \details Keeps one extra character available so the final quote and null terminator can be appended.
236 bool bufAppendMsgChar(wchar_t* buf, size_t& pos, unsigned int uChr) noexcept
237 {
238 if(pos + 2u >= cmdLineMax) return false;
239
240 if(uChr == '\n') buf[pos++] = L'\n';
241 else if(uChr == '\t') buf[pos++] = L' ';
242 else if(uChr < 32u || uChr > 126u) buf[pos++] = L'?';
243 else if(uChr == '"') buf[pos++] = L'\'';
244 else if(uChr == '\\') buf[pos++] = L'/';
245 else buf[pos++] = static_cast<wchar_t>(uChr);
246
247 buf[pos] = L'\0';
248 return true;
249 }

251 /// \brief Appends a null-terminated ASCII message to the command line.
252 void bufAppendMsg(wchar_t* buf, size_t& pos, const char* text) noexcept
253 {
254 if(!text) return;
255 while(*text) {
256 if(!bufAppendMsgChar(buf, pos, static_cast<unsigned char>(*text)))
257 break;
258 ++text;
259 }
260 }

271 /// \brief Finalizes the command line and launches npiMsgBox.exe.
272 /// \param buf Writable wide-character buffer construction.
273 /// \param pos Index of the next available element in `buf`.
274 /// \param type 0 = information, 1 = warning, 2 = stop/error.
275 void launch(wchar_t* buf, size_t pos, int type = 0) noexcept
276 {
277 if(pos + 5u >= cmdLineMax) return;
278
279 // Close the message argument, then append the severity argument.
280 buf[pos++] = L'"';
281 buf[pos++] = L' ';
282 buf[pos++] = L'"';
283 if(type == 0) buf[pos++] = L'0';
284 else if(type == 1) buf[pos++] = L'1';
285 else if(type == 2) buf[pos++] = L'2';
286 buf[pos++] = L'"';
287 buf[pos] = L'\0';
288
289 STARTUPINFOW si{};
290 si.cb = sizeof(si);
291
292 PROCESS_INFORMATION pi{};
293
294 if(::CreateProcessW(
295 exePathBuf, // lpApplicationName: full path to npiMsgBox.exe
296 buf, // lpCommandLine: "exe" "title" "message" "type"
297 nullptr,
298 nullptr,
299 FALSE,
300 CREATE_NO_WINDOW,
301 nullptr,
302 nullptr,
303 &si,
304 &pi)) {
305 // Allows the helper process to bring its message box to the foreground.
306 ::AllowSetForegroundWindow(pi.dwProcessId);
307
308 ::CloseHandle(pi.hThread);
309 ::CloseHandle(pi.hProcess);
310 }
311 }

313 void claimStopOwnership() noexcept
314 {
315 const std::uint32_t selfThreadId = static_cast<std::uint32_t>(::GetCurrentThreadId());
316 std::uint32_t expectedThreadId{0};
317 if(stopCallOwnerThreadId.compare_exchange_strong(
318 expectedThreadId,
319 selfThreadId,
320 std::memory_order_acq_rel,
321 std::memory_order_acquire))
322 return;
323
324 if(expectedThreadId == selfThreadId) {
325 ::TerminateProcess(::GetCurrentProcess(), EXIT_FAILURE);
326 std::abort(); // fallback
327 }
328
329 ::Sleep(5000);
330
331 ::TerminateProcess(::GetCurrentProcess(), EXIT_FAILURE);
332 std::abort(); // fallback
333 }
