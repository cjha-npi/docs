42QColor npi::wid::ColorPicker::GetColor_(const QColor& initial, QWidget* parent, const QString& title, const Config& config)
43{
44 QDialog dialog{parent};
45 dialog.setWindowTitle(title);
46
47 auto* layout = new QVBoxLayout{&dialog};
48 auto* picker = new ColorPicker{parent, true, initial, config, Design{}};
49 auto* buttons = new QDialogButtonBox{
50 QDialogButtonBox::Ok | QDialogButtonBox::Cancel,
51 Qt::Horizontal,
52 &dialog
53 };
54
55 layout->addWidget(picker);
56 layout->addWidget(buttons);
57
58 // Start the dialog at the smallest size that fits its controls.
59 if(auto* lay = dialog.layout()) {
60 lay->activate();
61 dialog.resize(lay->minimumSize());
62 }
63
64 QObject::connect(buttons, &QDialogButtonBox::accepted, &dialog, &QDialog::accept);
65 QObject::connect(buttons, &QDialogButtonBox::rejected, &dialog, &QDialog::reject);
66
67 if(dialog.exec() != QDialog::Accepted)
68 return {};
69
70 return picker->clr_;
71}

163void npi::wid::ColorPicker::setColor_(const QColor& color, bool emitSignal)
164{
165 if(!color.isValid()) return;
166
167 // Avoid handling intermediate changes while updating the editors.
168 const auto slotBlocker{slotGate_.block_()};
169
170 if(cfg_.spec_ == Spec::RGB) {
171 ch1_.spn_->setValue(color.red());
172 ch2_.spn_->setValue(color.green());
173 ch3_.spn_->setValue(color.blue());
174 }
175 else if(cfg_.spec_ == Spec::HSL) {
176 ch1_.spn_->setValue(color.hslHue());
177 ch2_.spn_->setValue(color.hslSaturation());
178 ch3_.spn_->setValue(color.lightness());
179 }
180 else if(cfg_.spec_ == Spec::HSV) {
181 ch1_.spn_->setValue(color.hsvHue());
182 ch2_.spn_->setValue(color.hsvSaturation());
183 ch3_.spn_->setValue(color.value());
184 }
185 aCh_.spn_->setValue(color.alpha());
186
187 // Use the editor values so channel limits are respected.
188 updateColor_(emitSignal);
189}

968void npi::wid::ColorPicker::updateColor_(bool emitSignal)
969{
970 const int val1 = ch1_.spn_->value();
971 const int val2 = ch2_.spn_->value();
972 const int val3 = ch3_.spn_->value();
973 const int aVal = aCh_.spn_->value();
974
975 QColor color;
976 if(cfg_.spec_ == Spec::RGB)
977 color = QColor{val1, val2, val3, aVal};
978 else if(cfg_.spec_ == Spec::HSL)
979 color = QColor::fromHsl(val1, val2, val3, aVal);
980 else if(cfg_.spec_ == Spec::HSV)
981 color = QColor::fromHsv(val1, val2, val3, aVal);
982
983 if(!color.isValid()) return;
984
985 const bool changed{color.rgba() != clr_.rgba()};
986
987 // Preserve hue changes even when the displayed color stays the same.
988 clr_ = color;
989
990 if(changed) {
991 clrBtn_->setProperty(npi::gui::Props::ButtonColorDisplay_, clr_);
992 if(!dialog_ && emitSignal) emit colorChanged_(clr_);
993 }
