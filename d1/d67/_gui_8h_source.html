1/**
2 * \file Gui.h
3 * \brief Configuration and control of application's graphical user interface.
4 *
5 * \details This file contains the public interface for the application's
6 * graphical user interface. The application's graphical user interface
7 * provides a centralized application style with dark, light, and
8 * system-following themes.
9 *
10 * This file exposes functions for selecting and querying the current theme,
11 * determining the effective dark/light background, and retrieving
12 * GUI colors for the current or an explicitly selected background.
13 *
14 * This file also declares QObject dynamic properties used to request specialized
15 * widget behavior, including theme-dependent icons, password line edits, item-view
16 * widgets, square sizing, grouped widgets and buttons, custom QPushButton layout,
17 * color-display buttons, and wheel-event forwarding.
18 */

27/**
28 * \namespace npi::gui
29 * \brief Configuration and control of application's graphical user interface.
30 *
31 * \details Provides the public interface for querying and controlling the
32 * application GUI, including theme selection, theme colors and special
33 * widget dynamic properties.
34 *
35 * \par Initialization
36 * The GUI structure is initialized automatically through either
37 * `QApplication::exec()` or a call to any of the functions inside
38 * this namespace, whichever happens earlier, provided that
39 * the `QApplication` object exists and shutdown has not begun.
40 *
41 * The preferred approach is to call `QCoreApplication::processEvents();`
42 * immediately after `QApplication` object construction in `main()`,
43 * before creating any widgets. This processes deferred initialization
44 * for the GUI and other application structures. This way GUI is initialized
45 * before any widgets are created.
46 *
47 * \note
48 * - Do not extend this namespace beyond this `Gui.h` file.
49 * - Usage of this namespace should be only through GUI thread.
50 * - This namespace requires `QApplication` instance to exist.
51 */

54 /// \brief Dynamic property names for specialized GUI behavior.
55 /// \details Set these properties using `QObject::setProperty()` or
56 /// as dynamic properties in Qt Designer using property string names.
57 /// Each property documents its expected value type and supported widgets.
58 ///
59 /// \note Unless otherwise stated, set properties before the widget is
60 /// polished by the application style. If the widget is already polished,
61 /// call `widget->style()->polish(widget)` after setting the property.
62 ///
63 /// \warning Removing a property does not necessarily undo its configuration.
64 /// In particular, group-base and item-view configuration may permanently
65 /// alter widget backgrounds, sizing or button grouping. Polishing again
66 /// does not restore the original configuration.
67 struct Props final {
68 /// \brief \b QString - Selects separator borders for backgroundless controls.
69 /// \details Use T (top), R (right), B (bottom) and L (left), in any
70 /// combination and case. For example, "TB" draws top and bottom separators.
71 /// Separator ends are inset from the corners.
72 ///
73 /// \note
74 /// - Applies to `QPushButton`, `QToolButton`, `QComboBox`, `QLineEdit` and
75 /// `QAbstractSpinBox` when their backgrounds are suppressed by group-base
76 /// or item-view configuration.
77 /// - On a `QFrame` with `HLine` or `VLine` shape, the property's presence
78 /// requests inset line ends; its string value is ignored.
79 /// - Supports runtime changes without polishing. Call `QWidget::update()`
80 /// after changing the property to schedule a repaint.
81 static constexpr const char* SepBorders_{"npi_SepBorders"};
82
83 /// \brief \b bool - Forwards wheel input to the widget's parent.
84 /// \details When enabled and a parent widget exists, wheel events are
85 /// forwarded to that parent instead of being handled by this widget.
86 /// Useful for preventing wheel input from changing a control's value
87 /// while allowing its surrounding container to scroll.
88 ///
89 /// \note
90 /// - If no parent widget exists, normal wheel handling is preserved.
91 /// - The original event is consumed even if the parent ignores it.
92 static constexpr const char* WheelToParent_{"npi_WheelToParent"};
93
94 /// \brief \b QString - Resource name for a theme-dependent icon or label pixmap.
95 /// \details Specify the name without a path or extension. Resource lookup uses
96 /// `:/icons/dark/<name>.png` or `:/icons/light/<name>.png`, according to the
97 /// effective background. If unavailable, `:/icons/<name>.png` is used.
98 ///
99 /// \note
100 /// - Applies to `QAbstractButton`, `QAction` and `QLabel`.
101 /// - For buttons and actions, this supplies the `QIcon::Off` state.
102 /// Use `IconOnName_` for the `QIcon::On` state.
103 /// - Icons are refreshed when the effective dark/light background changes.
104 /// - Set action properties before adding the action to its GUI control.
105 ///
106 /// \warning If neither resource exists, a warning is logged. Buttons and
107 /// actions fall back to the corresponding Normal-mode pixmaps from the icon
108 /// stored on their first dynamic-icon update. Labels retain their current pixmap.
109 static constexpr const char* IconName_{"npi_IconName"};
110
111 /// \brief \b QString - Resource name for a theme-dependent `QIcon::On` state.
112 /// \details Uses the same resource lookup and fallback behavior as `IconName_`.
113 /// The Off and On resources are resolved independently.
114 ///
115 /// \note
116 /// - Applies to `QAbstractButton` and `QAction` only.
117 /// - This property supplies an icon state; it does not make the button
118 /// or action checkable.
119 static constexpr const char* IconOnName_{"npi_IconOnName"};
120
121 /// \brief \b bool - Configures a `QLineEdit` for password entry.
122 /// \details Sets `QLineEdit::Password` echo mode, enables the clear button
123 /// and adds a leading action that toggles between hidden and visible text.
124 ///
125 /// \note Initialization occurs once. Repeated polishing does not add
126 /// duplicate actions or reset the current password visibility.
127 ///
128 /// \warning Replaces the existing echo mode. Disabling or removing this
129 /// property does not undo the configuration or remove the visibility action.
130 static constexpr const char* PasswordLineEdit_{"npi_PasswordLineEdit"};
131
132 /// \brief \b bool - Requests a width equal to the widget's calculated height.
133 /// \details Adjusts the style's size hint to request square sizing.
134 /// The width remains constrained by `minimumWidth()` and `maximumWidth()`.
135 ///
136 /// \note
137 /// - Menu-free, icon-only `QPushButton` controls and `QToolButton` controls
138 /// displaying only an icon or arrow request square sizing automatically.
139 /// Tool buttons directly inside `QLineEdit` are excluded from this automatic
140 /// behavior.
141 /// - This affects the size hint, not a fixed size; the layout may assign
142 /// different dimensions.
143 ///
144 /// \warning Does not apply to `QLabel` or widgets configured as item-view widgets.
145 static constexpr const char* WidthSameAsHeight_{"npi_WidthSameAsHeight"};
146
147 /// \brief \b bool - Marks a widget for use inside an item view.
148 /// \details Configures the marked widget and its descendants for item-view
149 /// background handling and sizing. Children added later are handled
150 /// automatically. Configuration does not extend into child windows.
151 ///
152 /// \note
153 /// - Set this on the root widget before calling `setCellWidget()`,
154 /// `setItemWidget()` or `setIndexWidget()`, and before the widget is polished.
155 /// - Descendants do not need this property.
156 /// - Item-view background handling takes precedence over group-base styling.
157 ///
158 /// \warning Removing this property does not restore the original configuration.
159 /// Configured widgets are not intended to be reparented and reused as
160 /// ordinary widgets outside an item view.
161 static constexpr const char* ItemViewWidget_{"npi_ItemViewWidget"};
162
163 /// \brief \b bool - Provides a sunken background for a group of child widgets.
164 /// \details Draws a rounded surface using `Surfaces::sunken_`. Descendant
165 /// backgrounds are suppressed unless they have `SunkenBase_` or `RaisedBase_`
166 /// enabled. Children added later are handled automatically. Configuration
167 /// does not extend into child windows.
168 ///
169 /// \note
170 /// - Item-view background handling takes precedence.
171 /// - If both `SunkenBase_` and `RaisedBase_` are enabled, `SunkenBase_` takes precedence.
172 /// - Does not group buttons. Enable `ButtonGroupBase_` on the same widget
173 /// for exclusive button grouping with this background.
174 /// - After changing this property at runtime, manually polish the base widget.
175 ///
176 /// \warning Removing this property does not restore the original configuration.
177 /// Configured descendants are not intended to be reparented and reused as
178 /// ordinary widgets outside the group.
179 static constexpr const char* SunkenBase_{"npi_SunkenBase"};
180
181 /// \brief \b bool - Provides a raised background for a group of child widgets.
182 /// \details Draws a rounded surface using `Surfaces::raised_`. Descendant
183 /// backgrounds are suppressed unless they have `SunkenBase_` or `RaisedBase_`
184 /// enabled. Children added later are handled automatically. Configuration
185 /// does not extend into child windows.
186 ///
187 /// \note
188 /// - Item-view background handling takes precedence.
189 /// - If both `RaisedBase_` and `SunkenBase_` are enabled, `SunkenBase_` takes precedence.
190 /// - Does not group buttons. Enable `ButtonGroupBase_` on the same widget
191 /// for exclusive button grouping with this background.
192 /// - After changing this property at runtime, manually polish the base widget.
193 ///
194 /// \warning Removing this property does not restore the original configuration.
195 /// Configured descendants are not intended to be reparented and reused as
196 /// ordinary widgets outside the group.
197 static constexpr const char* RaisedBase_{"npi_RaisedBase"};
198
199 /// \brief \b bool - Provides a shared background with exclusive button grouping.
200 /// \details Immediate `QPushButton` and `QToolButton` children are made checkable
201 /// and added to an exclusive `QButtonGroup`. An existing direct-child group is
202 /// reused; otherwise one is created. Children added later are handled automatically.
203 ///
204 /// Descendant backgrounds are suppressed unless they have `SunkenBase_` or
205 /// `RaisedBase_` enabled. Configuration does not extend into child windows.
206 ///
207 /// \note
208 /// - Uses a rounded `Surfaces::raised_` background unless `SunkenBase_` is enabled.
209 /// - When nested inside another group base, its background is suppressed unless
210 /// `SunkenBase_` or `RaisedBase_` is also enabled.
211 /// - Item-view background handling takes precedence.
212 /// - After changing this property at runtime, manually polish the base widget.
213 ///
214 /// \warning Removing this property does not undo button grouping, checkability
215 /// or background configuration. Configured descendants are not intended to be
216 /// reparented and reused as ordinary widgets outside the group.
217 static constexpr const char* ButtonGroupBase_{"npi_ButtonGroupBase"};
218
219 /// \brief \b bool - Places a `QPushButton`'s icon to the right of its text.
220 /// \note Only affects buttons containing both text and an icon.
221 static constexpr const char* ButtonRightIcon_{"npi_ButtonRightIcon"};
222
223 /// \brief \b bool - Aligns `QPushButton` text toward its icon instead of centering it.
224 /// \note Only affects buttons containing both text and an icon. Works with
225 /// `ButtonRightIcon_` for icons placed on either side.
226 static constexpr const char* ButtonTextNearIcon_{"npi_ButtonTextNearIcon"};
227
228 /// \brief \b QColor - Makes a `QPushButton` display the specified color.
229 /// \details Replaces normal button visuals with a color preview over a
230 /// white and gray checkerboard to show transparency. Text, icons and
231 /// pressed, focus and default-button indicators are not drawn.
232 /// When disabled, the color is shown in grayscale while preserving its alpha.
233 ///
234 /// \note Set the property before polishing. Its color value can then be
235 /// changed at runtime, with automatic repainting.
236 ///
237 /// \warning Changes the focus policy to `Qt::NoFocus` but does not disable
238 /// mouse interaction or the `clicked()` signal. Removing the property does
239 /// not restore the previous focus policy.
240 static constexpr const char* ButtonColorDisplay_{"npi_ButtonColorDisplay"};
241 };

257 /// \brief Returns the configured application theme.
258 /// \details Returns Theme::System when following Windows,
259 /// rather than the resolved dark/light theme. Use
260 /// isDarkBackground() for the effective background.
261 /// \return Theme::Dark, Theme::Light or Theme::System.
262 ///
263 /// \note Returns Theme::System if QApplication does not
264 /// exist or shutdown has begun.

267 /// \brief Selects the application theme.
268 /// \param theme Dark, Light or System. System follows the Windows application theme.
269 /// \details Updates the application appearance when the effective dark/light
270 /// background changes, including palettes and registered theme-dependent icons.
271 ///
272 /// \note Does nothing if QApplication does not exist or shutdown has begun.
273 /// Such requests are not retained for later initialization.
274 void setTheme(Theme theme);

276 /// \brief Returns whether the effective application background is dark.

278 /// \return true for a dark background; false for a light background.
