29/**
30 * \namespace npi::gui
31 * \brief Configuration and control of the application's graphical user interface.
32 *
33 * \details Provides the public interface for querying and controlling the
34 * application GUI, including theme selection, theme colors and special
35 * widget dynamic properties.
36 *
37 * \par Initialization
38 * GUI initialization is scheduled during application startup and runs when
39 * events are first processed, through `QApplication::exec()` or an explicit
40 * `QCoreApplication::processEvents()` call. Calling any function declared in
41 * this namespace also initializes the GUI if initialization is permitted.
42 * Initialization requires a fully constructed `QApplication` and does not
43 * occur after application shutdown has begun.
44 *
45 * The preferred application startup sequence is to call
46 * `QCoreApplication::processEvents()` immediately after constructing
47 * `QApplication` in `main()`, before creating widgets. This processes the
48 * deferred GUI initialization and any other pending startup events.
49 *
50 * \note
51 * - Use this interface from the GUI thread.
52 * - Normal use requires a `QApplication`. Individual functions document their
53 * fallback behavior when no application exists or shutdown has begun.
54 *
55 * \warning Do not extend this namespace beyond `Gui.h` file.
56 */

179 /// \details Presence enables display mode, including a value of false:

216 /// \brief Connects a callback to effective background changes.
217 /// \note The callback runs after the background's palette and dynamic icons are updated.
218 /// The connection is removed when the context is destroyed.
219 /// \return An invalid connection if context is null, callback is empty,
220 /// or GUI initialization is unavailable.

223 /// \brief Connects a member function to selected-theme changes.
224 /// \details The signal is emitted after the background and shared actions are updated.
225 /// \note The connection is removed automatically when the receiver is destroyed.
226 /// \return The connection, or an invalid connection if the receiver or method
227 /// is null, or GUI initialization is unavailable.
228 template <std::derived_from<QObject> T>
229 QMetaObject::Connection onThemeChanged(T* receiver, void (T::*method)(Theme))
230 {
231 if(!receiver || !method)
232 return {};
233
234 return onThemeChanged(
235 static_cast<QObject*>(receiver),
236 [receiver, method](Theme theme){
237 (receiver->*method)(theme);
238 });
239 }
240
241 /// \brief Connects a member function to effective background changes.
242 /// \details The signal provides true for a dark background and false for a
243 /// light background. It is emitted after palettes and dynamic icons are updated.
244 /// \note The connection is removed automatically when the receiver is destroyed.
245 /// \return The connection, or an invalid connection if the receiver or method
246 /// is null, or GUI initialization is unavailable.
247 template <std::derived_from<QObject> T>
248 QMetaObject::Connection onBackgroundChanged(T* receiver, void (T::*method)(bool))
249 {
250 if(!receiver || !method)
251 return {};
252
253 return onBackgroundChanged(
254 static_cast<QObject*>(receiver),
255 [receiver, method](bool dark){
256 (receiver->*method)(dark);
257 });
258 }
