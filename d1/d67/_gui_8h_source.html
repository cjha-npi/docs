23/**
24 * \namespace npi::gui
25 * \brief Configuration and control of the application's graphical user interface.
26 *
27 * \details Provides the public interface for querying and controlling the
28 * application GUI, including theme selection, theme colors and special
29 * widget dynamic properties.
30 *
31 * \par Initialization
32 * GUI initialization is scheduled during application startup and runs when
33 * events are first processed, through `QApplication::exec()` or an explicit
34 * `QCoreApplication::processEvents()` call. Calling any function declared in
35 * this namespace also initializes the GUI if initialization is permitted.
36 * Initialization requires a fully constructed `QApplication` and does not
37 * occur after application shutdown has begun.
38 *
39 * The preferred application startup sequence is to call
40 * `QCoreApplication::processEvents()` immediately after constructing
41 * `QApplication` in `main()`, before creating widgets. This processes the
42 * deferred GUI initialization and any other pending startup events.
43 *
44 * \note
45 * - Use this interface from the GUI thread.
46 * - Normal use requires a `QApplication`. Individual functions document their
47 * fallback behavior when no application exists or shutdown has begun.
48 *
49 * \warning Do not extend this namespace beyond `Gui.h` file.
50 */

113 /// \details Adjusts supported size hints within minimum/maximum width

122 /// \brief \b bool - Configures an embedded item-view widget and its descendants.
123 /// \details Suppresses backgrounds and changes Fixed vertical size policies

125 ///
126 /// \note Set on the root before `setCellWidget()`, `setItemWidget()` or `setIndexWidget()`.
127 /// If set later, repolish the owning item view, not just the embedded widget.

130 /// widgets for reuse outside the item view.
131 static constexpr const char* ItemViewWidget_{"npi_ItemViewWidget"};

133 /// \brief \b bool - Provides a rounded Surfaces::sunken_ group background.
134 /// \details Suppresses descendant backgrounds unless they have their own
135 /// `npi::gui::Props::SunkenBase_` or `npi::gui::Props::RaisedBase_`.
136 /// Takes precedence over `npi::gui::Props::RaisedBase_`. Does not enable
137 /// button grouping; intended primarily for text-entry controls.
138 ///
139 /// \warning Avoid checkable push/tool buttons because checked-state contrast
140 /// can be poor. Removal does not restore the original configuration.
141 static constexpr const char* SunkenBase_{"npi_SunkenBase"};
142
143 /// \brief \b bool - Provides a rounded Surfaces::raised_ group background.
144 /// \details Suppresses descendant backgrounds unless they have their own
145 /// `npi::gui::Props::SunkenBase_` or `npi::gui::Props::RaisedBase_`.
146 /// `npi::gui::Props::SunkenBase_` takes precedence if both are set.
147 /// Does not enable button grouping; combine with
148 /// `npi::gui::Props::ButtonGroupBase_` if needed.
149 ///
150 /// \warning Removal does not restore the original configuration.
151 static constexpr const char* RaisedBase_{"npi_RaisedBase"};
152
153 /// \brief \b bool - Provides a shared background with exclusive button grouping.
154 /// \details Immediate `QPushButton` and `QToolButton` children become checkable and
155 /// join an exclusive `QButtonGroup`. An existing direct-child group is reused.
156 /// Tool buttons with checkable default actions in an exclusive `QActionGroup`
157 /// are excluded. Uses the raised background unless `npi::gui::Props::SunkenBase_`
158 /// is enabled; nesting follows the group-base rules in GUI Systems documentation.
159 ///
160 /// \note No initial selection is imposed. Repolish the base to reevaluate grouping.
161 /// See GUI Systems documentation for action assignment and runtime restrictions.
162 ///
163 /// \warning Reused groups become managed and exclusive. Removing the property
164 /// does not undo grouping, checkability or background configuration.
165 static constexpr const char* ButtonGroupBase_{"npi_ButtonGroupBase"};
166
167 /// \brief \b bool - Aligns QPushButton text toward its icon instead of centering it.
168 ///
169 /// \note Only affects buttons containing both text and an icon.
170 static constexpr const char* ButtonTextNearIcon_{"npi_ButtonTextNearIcon"};
171
172 /// \brief \b QVariant - Overrides QPushButton and QToolButton background drawing.
173 /// \details Presence enables display mode, including a value of false:
174 /// - Convertible to valid `QColor`: fills over a transparency checkerboard;
175 /// disabled colors become grayscale with alpha preserved.
176 /// - Otherwise: no fill; an iconless, textless button gets a thin raised-color border.
177 /// Content, menu indicators and `npi::gui::Props::SepBorders_` remain; normal background-state
178 /// effects and focus indicators are omitted. Sizing is unchanged.
179 ///
180 /// \note Set before polishing. Subsequent value changes/removal repaint automatically.
181 /// Removal restores normal drawing. Qt-managed internal buttons can be exempt.
182 ///
183 /// \warning
184 /// - Polishing applies `Qt::NoFocus` without disabling button interaction.
185 /// - Removing the property does not restore the previous focus policy.
186 static constexpr const char* ButtonDisplay_{"npi_ButtonDisplay"};

214 /// \brief Returns the shared checkable action for selecting a theme.
215 /// \details Its exclusive checked state follows `theme()`, not the effective background.
216 /// Use `QToolButton::setDefaultAction()` or add to menus/toolbars; no extra
217 /// theme-setting connection is needed. All users share the action's properties.
218 /// \return Theme `QAction` pointer; nullptr for an invalid theme,
219 /// absent `QApplication` or during shutdown.
220 /// \warning The returned `QAction*` is owned by the GUI style. Do not delete,
221 /// reparent, regroup or change checkability/checked state. Select themes through
222 /// `setTheme()` or `QAction::trigger()`. See GUI Systems documentation for use
223 /// inside `npi::gui::Props::ButtonGroupBase_`.
224 QAction* themeAction(Theme theme);
