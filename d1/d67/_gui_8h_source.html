1/**
2 * \file Gui.h
3 * \brief Declares runtime configuration and public controls for the graphical user interface.
4 *
5 * \details
6 * The GUI system provides a centralized application style with dark, light, and
7 * system-following themes. This file exposes functions for selecting and querying
8 * the current theme, determining the effective dark/light background, and retrieving
9 * GUI colors for the current or an explicitly selected background.
10 *
11 * This file also declares QObject dynamic properties used to request specialized
12 * widget behavior, including theme-dependent icons, password line edits, item-view
13 * widgets, square sizing, grouped widgets and buttons, custom QPushButton layout,
14 * color-display buttons, and wheel-event forwarding.
15 *
16 * Unless otherwise documented, GUI properties are intended to be set before the
17 * widget is polished by the application style. Properties that require initialization
18 * after being set must be applied through a subsequent manual polish when that timing
19 * cannot be guaranteed.
20 */

44 /**
45 * \struct Props
46 * \brief QObject dynamic properties providing special GUI functionality.
47 *
48 * \note
49 * - Unless otherwise stated, these properties should be set before
50 * the widget is polished by the application style. If that cannot be
51 * guaranteed, manually polish the widget after setting the property.
52 * - `ButtonColorDisplay_` supports runtime color changes with automatic repainting.
53 * - The group-base properties may be changed at runtime. Manually polish the
54 * affected widget after changing `SunkenBase_`, `RaisedBase_` or `ButtonGroupBase_`.
55 * - Removing `ButtonGroupBase_` does not undo button grouping or checkability.
56 * - Removing all group-base properties or `ItemViewWidget_` does not restore
57 * the original configuration of the widget or its descendants. Manual polish
58 * does not perform that restoration.
59 */

66 /// \brief \b QString - Name of the theme-dependent icon without path or extension.
67 /// \details The active theme-specific resource is searched first:
68 /// `:/icons/dark/<name>.png` for the dark theme or
69 /// `:/icons/light/<name>.png` for the light theme. If that resource
70 /// does not exist, `:/icons/<name>.png` is used as the fallback.
71 /// \note
72 /// - If neither resource exists, a warning is logged. Buttons and actions
73 /// use the stored original icon's corresponding state as fallback;
74 /// labels retain their existing pixmap.
75 /// - Only applies to QAbstractButton, QLabel and QAction.
76 static constexpr const char* IconName_{"npi_IconName"};
77
78 /// \brief \b QString - Name of the theme-dependent ON-state icon without path or extension.
79 /// \details Uses the same theme-specific and fallback resource lookup as IconName_.
80 /// \note Applies only to QAbstractButton and QAction.
81 static constexpr const char* IconOnName_{"npi_IconOnName"};
82
83 /// \brief \b bool - Configures a QLineEdit for password entry.
84 /// \details Sets password echo mode, enables the clear button and
85 /// adds a leading action to show/hide the password.
86 /// \warning Replaces the QLineEdit's existing echo mode.
87 static constexpr const char* PasswordLineEdit_{"npi_PasswordLineEdit"};
88
89 /// \brief \b bool - Marks a widget intentionally installed as an item-view widget.
90 /// \details The marked root widget and its descendants are configured for correct
91 /// background handling, sizing and runtime child handling inside an item view.
92 /// \note Set this on the root widget before using setCellWidget(), setItemWidget()
93 /// or setIndexWidget(), and before the widget is polished, preferably immediately
94 /// after creating the widget. Descendants are handled automatically.
95 /// \warning A widget configured as an item-view widget, or one of its configured
96 /// descendants, is not intended to be removed and reparented for reuse as an ordinary widget.
97 static constexpr const char* ItemViewWidget_{"npi_ItemViewWidget"};
98
99 /// \brief \b bool - Requests a widget width equal to its calculated height.
100 /// \note
101 /// - Menu-free icon-only QPushButton controls and QToolButton controls
102 /// displaying only an icon or arrow request square sizing automatically.
103 /// Tool buttons directly inside QLineEdit are excluded.
104 /// - The resulting width is constrained by the widget's minimumWidth()
105 /// and maximumWidth().
106 /// \warning Does not apply to widgets inside QAbstractItemView or to QLabel.
107 static constexpr const char* WidthSameAsHeight_{"npi_WidthSameAsHeight"};
108
109 /// \brief \b bool - Makes a widget a sunken visual base for a group of child widgets.
110 /// \details The base is drawn as a single rounded surface using
111 /// `Surfaces::sunken_`. Descendant widgets without `SunkenBase_`
112 /// or `RaisedBase_` enabled are configured not to draw their
113 /// own background so the group appears as one control.
114 /// \note
115 /// - Item-view configuration takes precedence over this property.
116 /// - If both `RaisedBase_` and `SunkenBase_` are enabled, `SunkenBase_` takes precedence.
117 /// - Widgets may be added and removed dynamically at runtime.
118 /// - This property does not configure button grouping. Enable
119 /// `ButtonGroupBase_` on the same widget for automatic button grouping;
120 /// this property overrides its default background color.
121 /// \warning Widgets configured as part of this group are not intended to be
122 /// removed, reparented and reused as ordinary widgets.
123 static constexpr const char* SunkenBase_{"npi_SunkenBase"};
124
125 /// \brief \b bool - Makes a widget a raised visual base for a group of child widgets.
126 /// \details The base is drawn as a single rounded surface using
127 /// `Surfaces::raised_`. Descendant widgets without `SunkenBase_`
128 /// or `RaisedBase_` enabled are configured not to draw their
129 /// own background so the group appears as one control.
130 /// \note
131 /// - Item-view configuration takes precedence over this property.
132 /// - If both `RaisedBase_` and `SunkenBase_` are enabled, `SunkenBase_` takes precedence.
133 /// - Widgets may be added and removed dynamically at runtime.
134 /// - This property does not configure button grouping. Enable
135 /// `ButtonGroupBase_` on the same widget for automatic button grouping;
136 /// this property overrides its default background color.
137 /// \warning Widgets configured as part of this group are not intended to be
138 /// removed, reparented and reused as ordinary widgets.
139 static constexpr const char* RaisedBase_{"npi_RaisedBase"};
140
141 /// \brief \b bool - Configures a shared visual base with exclusive button grouping.
142 /// \details Immediate `QPushButton` and `QToolButton` children are made checkable
143 /// and added to one exclusive `QButtonGroup`. An existing direct-child group is
144 /// reused; otherwise one is created automatically.
145 /// Descendant backgrounds are suppressed except on widgets with `SunkenBase_`
146 /// or `RaisedBase_` enabled.
147 /// \note
148 /// - The base uses a rounded `Surfaces::raised_` background unless overridden
149 /// by `SunkenBase_` or `RaisedBase_` on the same widget.
150 /// - When nested inside another group base, its own background is suppressed
151 /// unless `SunkenBase_` or `RaisedBase_` is enabled.
152 /// - Item-view configuration takes precedence over background drawing.
153 /// - Widgets may be added and removed dynamically at runtime.
154 /// \warning Widgets configured as part of this group are not intended to be
155 /// removed, reparented and reused as ordinary widgets.
156 static constexpr const char* ButtonGroupBase_{"npi_ButtonGroupBase"};
157
158 /// \brief \b bool - Places a QPushButton icon on the right side of the button.
159 /// \note Only affects buttons containing both text and an icon.
160 static constexpr const char* ButtonRightIcon_{"npi_ButtonRightIcon"};
161
162 /// \brief \b bool - Aligns QPushButton text toward its icon instead of centering it.
163 /// \note Only affects buttons containing both text and an icon and works with
164 /// `ButtonRightIcon_` for icons placed on either side.
165 static constexpr const char* ButtonTextNearIcon_{"npi_ButtonTextNearIcon"};
166
167 /// \brief \b QColor - Makes a QPushButton display the specified color.
168 /// \details The button is drawn as a color display rather than with normal button
169 /// visuals: text, icon and pressed/focus/default visual states are not drawn.
170 /// The supplied color is composited over a white and gray checkerboard to show
171 /// transparency. When disabled, the color is converted to grayscale while
172 /// preserving its alpha component before compositing.
173 /// \note The property should be present before polish, but its QColor value may
174 /// be changed dynamically afterward and the button will repaint automatically.
175 /// \warning The button's focus policy is changed to Qt::NoFocus.
176 static constexpr const char* ButtonColorDisplay_{"npi_ButtonColorDisplay"};
177 };

226 /// \brief Selects a black or white text color for the specified background.
227 /// \details Uses a weighted RGB brightness heuristic, returning white for
228 /// darker backgrounds and black for lighter backgrounds. This does not
229 /// guarantee a particular contrast ratio.
230 /// \param background A valid background color. For a translucent background,

232 /// \return White if the weighted brightness is below 128; otherwise black.
233 /// \note The background's alpha component is ignored.
